20 /// Tag byte that starts every framed record (see ContainerRecordOutputArchive::close).
21 ///
22 /// A record larger than the RMI INT_MAX message limit cannot be sent through
23 /// `WorldContainer::replace` in one piece, so it is split: the record at `key`
24 /// becomes a header `[RECORD_CHUNKED][total][nchunk]` and the payload is stored as
25 /// `nchunk` records `[RECORD_CHUNK][key][i][slice]` at `container_record_chunk_key(key, i)`.
26 /// Any other record is `[RECORD_INLINE][payload]`, so the tag alone tells the reader
27 /// which it is.
28 ///
29 /// Chunk keys share the container's key space, so a chunk can still collide with an
30 /// unrelated record. That is unlikely (64-bit keys) but not prevented: ruling it out
31 /// would need an existence check on the remote owner, and WorldContainer's only remote
32 /// lookup, find(), copies the value, which may itself exceed 2 GiB. A collision is
33 /// detected instead: each chunk names its record and index, so the reader throws
34 /// rather than returning another record's bytes.
35 ///
36 /// Records are write-once: a key holding a chunked record must be freed with
37 /// erase_container_record() before it is written again, or the old chunks stay
38 /// allocated (not checked, for the same reason). The cloud never rewrites a key.
39 ///
40 /// An unframed record (`framed=false`) is the bare payload, never chunked: for
41 /// containers whose bytes are read directly rather than through
42 /// ContainerRecordInputArchive (the cloud's batch container, which has its own
43 /// chunked transport). Framed and unframed records must not share a container.
44 enum ContainerRecordTag : unsigned char {
45 RECORD_INLINE = 0x01,
46 RECORD_CHUNKED = 0x02,
47 RECORD_CHUNK = 0x03
48 };
49
50 /// Largest payload sent through a single WorldContainer::replace.
51 ///
52 /// The worldrmi assert is on the *total* message size (serialized key + value +
53 /// AM header), so we leave 1 MiB of headroom below INT_MAX for that overhead.
54 static constexpr std::size_t CONTAINER_RECORD_MAX_CHUNK =
55 static_cast<std::size_t>(std::numeric_limits<int>::max()) - (std::size_t(1) << 20);
56
57 /// Key of chunk `ichunk` of the chunked record at `key`: hashed, not `key+1, key+2, …`,
58 /// so a chunk cannot land on a neighbouring record whose key a caller chose sequentially.
59 ///
60 /// A full 64-bit mix (splitmix64's finalizer, a bijection), not worldhash's hash_value,
61 /// which folds a long to 32 bits: two records whose keys agreed in those 32 bits would
62 /// get identical chunk keys.
63 inline long container_record_chunk_key(const long key, const std::uint64_t ichunk) {
64 auto mix = [](std::uint64_t z) {
65 z = (z ^ (z >> 30)) * 0xbf58476d1ce4e5b9ull;
66 z = (z ^ (z >> 27)) * 0x94d049bb133111ebull;
67 return z ^ (z >> 31);
68 };
69 return long(mix(mix(std::uint64_t(key) ^ 0x4D41444E43484B53ull) + ichunk)); // "MADNCHKS"
70 }
71
72 /// size of a chunk's prefix `[RECORD_CHUNK][key][ichunk]`
73 static constexpr std::size_t CONTAINER_RECORD_CHUNK_PREFIX = 1 + sizeof(long) + sizeof(std::uint64_t);
74
75 /// parse a chunk header; false if `raw` is not one
76 ///
77 /// Not in a nested `detail` namespace: code that says `using namespace madness;
78 /// using namespace archive;` would then find `detail::` ambiguous.
79 inline bool parse_container_record_chunk_header(const std::vector<unsigned char>& raw,
80 std::uint64_t& total, std::uint64_t& nchunk) {
81 if (raw.size() != 1 + 2 * sizeof(std::uint64_t) or raw[0] != RECORD_CHUNKED) return false;
82 std::memcpy(&total, raw.data() + 1, sizeof(total));
83 std::memcpy(&nchunk, raw.data() + 1 + sizeof(total), sizeof(nchunk));
84 return true;
85 }
86
87 /// Erase a framed record and, if it is chunked, its chunks; erasing only `key` would
88 /// leave the chunks allocated.
89 inline void erase_container_record(WorldContainer<long, std::vector<unsigned char>>& dc,
90 const long key) {
91 auto it = dc.find(key).get();
92 if (it == dc.end()) return;
93 std::uint64_t total = 0, nchunk = 0;
94 const bool chunked = parse_container_record_chunk_header(it->second, total, nchunk);
95 dc.erase(key);
96 if (chunked)
97 for (std::uint64_t c = 0; c < nchunk; ++c) dc.erase(container_record_chunk_key(key, c));
98 }

155
156 // common case: the serialized record fits in a single RMI message
157 if (not framed or v.size() <= CONTAINER_RECORD_MAX_CHUNK) {
158 dc.replace(key, v);
159 return;
160 }
161
162 // oversized record: write a header at `key` and split the payload (which
163 // follows the inline tag at v[0]) into tagged chunks (see ContainerRecordTag).
164 const std::uint64_t total = v.size() - 1;
165 const std::size_t slice = CONTAINER_RECORD_MAX_CHUNK - CONTAINER_RECORD_CHUNK_PREFIX;
166 const std::uint64_t nchunk = (total + slice - 1) / slice;
167 std::vector<unsigned char> header(1 + 2 * sizeof(std::uint64_t));
168 header[0] = RECORD_CHUNKED;
169 std::memcpy(header.data() + 1, &total, sizeof(total));
170 std::memcpy(header.data() + 1 + sizeof(total), &nchunk, sizeof(nchunk));
171 dc.replace(key, header);
172
173 for (std::uint64_t c = 0; c < nchunk; ++c) {
174 const std::size_t start = 1 + c * slice;
175 const std::size_t end = std::min<std::size_t>(start + slice, v.size());
176 std::vector<unsigned char> chunk(CONTAINER_RECORD_CHUNK_PREFIX);
177 chunk[0] = RECORD_CHUNK;
178 std::memcpy(chunk.data() + 1, &key, sizeof(key));
179 std::memcpy(chunk.data() + 1 + sizeof(key), &c, sizeof(c));
180 chunk.insert(chunk.end(), v.begin() + start, v.begin() + end);
181 dc.replace(container_record_chunk_key(key, c), chunk);
182 }

193 // reassemble a chunked record: concatenate the payloads of its `nchunk` chunks
194 // (see ContainerRecordTag).
195 static std::vector<unsigned char>
196 load_chunks(const containerT& dc, const keyT& key, std::uint64_t total, std::uint64_t nchunk) {
197 std::vector<unsigned char> buf;
198 buf.reserve(total);
199 for (std::uint64_t c = 0; c < nchunk; ++c) {
200 containerT::const_iterator it = dc.find(container_record_chunk_key(key, c)).get();
201 MADNESS_CHECK_THROW(it != dc.end(), "ContainerRecordInputArchive: chunk record not found");
202 const std::vector<unsigned char>& chunk = it->second;
203 keyT parent = 0;
204 std::uint64_t index = 0;
205 const bool is_chunk = chunk.size() >= CONTAINER_RECORD_CHUNK_PREFIX and chunk[0] == RECORD_CHUNK;
206 if (is_chunk) {
207 std::memcpy(&parent, chunk.data() + 1, sizeof(parent));
208 std::memcpy(&index, chunk.data() + 1 + sizeof(parent), sizeof(index));
209 }
210 MADNESS_CHECK_THROW(is_chunk and parent == key and index == c,
211 "ContainerRecordInputArchive: chunk overwritten by another record (key collision)");
212 buf.insert(buf.end(), chunk.begin() + CONTAINER_RECORD_CHUNK_PREFIX, chunk.end());
213 }
214 MADNESS_CHECK_THROW(buf.size() == total,
215 "ContainerRecordInputArchive: chunked record size mismatch");
216 return buf;
217 }

222 containerT::const_iterator it = dc.find(key).get();
