4/// \file applyscreen.h
5/// \brief Destination screening of the apply of an integral operator.
6///
7/// The result of an apply is often multiplied with another function afterwards (the exchange operator:
8/// K f = sum_i ket_i * P(bra_i f)). A contribution to a box in which that multiplier is negligible is then
9/// never used, and need not be computed (Yanai, Fann, Gan, Harrison, Beylkin, JCP 121, 6680 (2004), eq. 17).
10/// A screen answers, per destination box, a bound on the multiplier's largest magnitude there (supbound.h).
11/// FunctionImpl::do_apply skips a block whose contribution times that bound, capped at one, is negligible,
12/// computes a kept block with its per-term tolerance loosened by the same factor, and accumulates it only
13/// if its norm times the factor is significant; a skipped block still counts as used by the shell loop.
14/// A skipped block leaves at most the block's share of the tolerance in the product function, since
15/// ||s du||_2 <= sup|s| ||du||_2. The tasks of one apply carry the screen (ApplyScreenHandle; apply_only,
16/// and the vector apply(world, op, f, screens)); without one, do_apply is unchanged.

36 /// The screen as the tasks of an apply carry it: shared ownership, so the screen lives as long as
37 /// any task of the apply does, and nothing is stored on the function. Empty means no screen. The
38 /// tasks stay on the rank that creates them (apply_only refuses apply_randomize), so the handle
39 /// does not travel.

57 /// Counters of the apply screen, process-wide: the blocks that passed the source test while a screen
58 /// was installed and the ones it skipped, and the sup bounds built for screens with the time it took.
59 /// Reset and reported by whoever installs the screens (the exchange operator, summed over the ranks);

65 /// one line on rank 0 with the counters summed over the ranks. Collective.
66 static void report(World& world) {
67 std::array<long, 4> n{long(blocks), long(skipped), long(sup_maps), long(sup_build_ns)};
68 world.gop.sum(n.data(), n.size());
69 if (world.rank() != 0) return;
70 printf(" apply screen: blocks %ld, skipped %ld (%.1f%%)", n[0], n[1],
71 n[0] > 0 ? 100.0 * double(n[1]) / double(n[0]) : 0.0);
72 if (n[2] > 0)
73 printf("; sup bounds of %ld multipliers built in %.2fs (%.1f ms each)", n[2], 1e-9 * double(n[3]),
74 1e-6 * double(n[3]) / double(n[2]));

76 }

79 /// The screen built from the multipliers' sup bounds (SupBoundMap, FunctionImpl::sup_bound_map): the
80 /// largest of their bounds in the box. A box none of the maps holds (not local to this rank, or without
81 /// coefficients) answers unbounded, which screens nothing.
82 template <std::size_t NDIM>
83 class SupBoundScreen {
84 std::vector<std::shared_ptr<const SupBoundMap<NDIM>>> sup_;
85 public:
86 static constexpr double unbounded() { return SupBoundMap<NDIM>::unbounded(); }
87
88 explicit SupBoundScreen(std::vector<std::shared_ptr<const SupBoundMap<NDIM>>> sup) : sup_(std::move(sup)) {
89 MADNESS_CHECK_THROW(not sup_.empty(), "SupBoundScreen: at least one multiplier");
90 for (const auto& s : sup_) MADNESS_CHECK_THROW(bool(s), "SupBoundScreen: a missing sup bound");
91 }
92
93 double operator()(const Key<NDIM>& key) const {
94 double s = 0.0;
95 for (const auto& m : sup_) s = std::max(s, (*m)(key));
96 return s;
97 }
98 };
