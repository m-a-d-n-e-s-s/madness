32/// \file population.h
33/// \brief atomic populations of occupied orbitals: Mulliken, Loewdin and intrinsic atomic orbitals (IAO)
34
35/// Everything here works on overlap matrices alone, so the same functions serve
36/// MRA orbitals (overlaps by quadrature) and Gaussian-basis orbitals (analytic
37/// overlaps). The inputs, for the occupied orbitals phi_i of one spin:
38/// - phi_i orthonormal, <phi_i|phi_j> = delta_ij, with occupations n_i
39/// - a basis chi_mu centred on the atoms; S = <chi|chi>, X = <chi|phi>
40/// - the atom each basis function sits on
41/// S and X must refer to the same functions chi (e.g. both from the same
42/// projection into MRA).
43

64/// electrons per atom from one spin's occupied orbitals, with the scheme's diagnostics
65struct SpinPopulation {
66 Tensor<double> electrons; ///< electrons on each atom
67 Tensor<double> completeness; ///< mulliken/lowdin: c_i = X_i^T S^-1 X_i, the share of phi_i inside the basis
68 double spilling = 0.0; ///< mulliken/lowdin: 1 - sum_i n_i c_i / sum_i n_i (Sanchez-Portal et al. 1995)
69 double min_w_eigenvalue = 0.0; ///< smallest eigenvalue of W = X^T S^-1 X (all schemes)
70 double max_norm_error = 0.0; ///< iao: max_i |sum_rho Q_rho,i^2 - 1|, zero when phi lies in the IAO span
71};

77/// Mulliken or Loewdin populations of the occupied space projected onto the basis
78
79/// The fit of phi_i onto the basis is C_i = S^-1 X_i; the fitted orbitals overlap as
80/// W = C^T S C = X^T S^-1 X, whose diagonal c_i <= 1 is each orbital's completeness. The
81/// fits are orthonormalized among themselves, Cbar = C W^-1/2 (the depolarized orbitals of
82/// the IAO construction), so the electrons sum to sum_i n_i and what the basis misses shows
83/// as the spilling instead:
84/// mulliken: N_mu = sum_i n_i (Cbar_i o S Cbar_i)_mu
85/// lowdin: N_mu = sum_i n_i ((S^1/2 Cbar_i)_mu)^2
86/// Both are invariant under rotations among equally occupied orbitals. (Renormalizing each
87/// fit by its own c_i also gives the sum rule, but not that invariance.)
88/// @param[in] scheme "mulliken" or "lowdin"
89SpinPopulation projection_population(const std::string& scheme, const Tensor<double>& S, const Tensor<double>& X,
90 const Tensor<double>& occ, const std::vector<int>& atom_of_bf, long natom);
91
92/// populations of the intrinsic atomic orbitals (Knizia, J. Chem. Theory Comput. 9, 4834 (2013))
93
94/// For a complete large basis (MRA) the IAOs follow from S and X of a minimal basis alone:
95/// W = X^T S^-1 X, D = S^-1 X W^-1/2 (the depolarized orbitals chi D)
96/// M = 1 - D D^T S, K = X^T (2 D D^T S - 1)
97/// A = chi M + phi K (the IAOs), <A|A> = M^T S M + M^T X K + K^T X^T M + K^T K, <A|phi> = M^T X + K^T
98/// Q = <A|A>^-1/2 <A|phi>, N_rho = sum_i n_i Q_rho,i^2
99/// The orbitals lie in the IAO span, so every column of Q has unit norm and the electrons
100/// sum to sum_i n_i to the precision of the overlaps. Throws when W is singular: the minimal
101/// basis then misses an orbital's atomic character.
102SpinPopulation iao_population(const Tensor<double>& S, const Tensor<double>& X, const Tensor<double>& occ,
103 const std::vector<int>& atom_of_bf, long natom);
104
105/// dispatch: "mulliken" and "lowdin" to projection_population, "iao" to iao_population
106SpinPopulation spin_population(const std::string& scheme, const Tensor<double>& S, const Tensor<double>& X,
107 const Tensor<double>& occ, const std::vector<int>& atom_of_bf, long natom);

137/// the population analysis of the population keys, for orbitals given in real space
138
139/// For each scheme of the `population` key: reads the basis set by name (population_basis
140/// for mulliken and lowdin, population_minbasis for iao), projects it at the current
141/// FunctionDefaults, forms S and X, and prints a table per scheme on rank 0 at print_level > 0.
142/// The occupied orbitals are those with a nonzero occupation, so explicit holes count, and
143/// the total charge is the nuclear charge minus their sum. The schemes assume orthonormal
144/// orbitals; after an SCF they are so only to its screening and truncation, so they are
145/// orthonormalized first (Loewdin, orthonormalize_symmetric). A failing scheme only warns:
146/// its entry in the result holds the error. moldft (SCF::population_analysis) and nemo
147/// (Nemo::population_analysis) call it after the final rung.
148class PopulationAnalysis {
149public:
150 /// the schemes and basis sets of the population keys; nothing is projected yet
151 PopulationAnalysis(World& world, const Molecule& molecule, const CalculationParameters& param);
152
153 bool requested() const; ///< false for `population none`
154
155 /// closed shell: both spins have the orbitals mo with the occupations occ
156 /// @return the results on every rank; empty when not requested()
157 nlohmann::json analyze(const std::vector<Function<double,3>>& mo, const Tensor<double>& occ) const;
158
159 /// open shell
160 nlohmann::json analyze(const std::vector<Function<double,3>>& amo, const Tensor<double>& aocc,
161 const std::vector<Function<double,3>>& bmo, const Tensor<double>& bocc) const;
162
163private:
164 World& world_;
165 std::shared_ptr<const Molecule> molecule_; ///< a copy: the results refer to this geometry
166 std::vector<std::string> schemes_; ///< the population key
167 std::string basis_, minbasis_; ///< population_basis, population_minbasis
168 int print_level_ = 0;
169
170 nlohmann::json run(const std::vector<Function<double,3>>& amo, const Tensor<double>& aocc,
171 const std::vector<Function<double,3>>& bmo, const Tensor<double>& bocc,
172 bool restricted) const;
173};
