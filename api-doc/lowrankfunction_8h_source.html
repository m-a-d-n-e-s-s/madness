139 /// grid with random points around the origin, with a Gaussian distribution
140 template<std::size_t NDIM>
141 class randomgrid : public gridbase {
142 public:
143 randomgrid(const double volume_element, const double radius, const Vector<double,NDIM> origin=Vector<double,NDIM>(0.0))
144 : gridbase(), origin(origin) {
145 this->volume_element=volume_element;
146 this->radius=radius;
147 }
148
149 std::vector<Vector<double,NDIM>> get_grid() const {
150 std::vector<Vector<double, NDIM>> grid;
151 long npoint_within_volume=volume()/volume_element;
152 if (do_print) print("npoint_within_volume",npoint_within_volume);
153
154 auto cell = FunctionDefaults<NDIM>::get_cell();
155 auto is_in_cell = [&cell](const Vector<double, NDIM>& r) {
156 for (size_t d = 0; d < NDIM; ++d) if (r[d] < cell(d, 0) or r[d] > cell(d, 1)) return false;
157 return true;
158 };
159 double rad=radius;
160 auto o=origin;
161 auto is_in_sphere = [&rad,&o](const Vector<double, NDIM>& r) {
162 return ((r-o).normf()<rad);
163 };
164
165 // set variance such that about 70% of all grid points sits within the radius
166 double variance=radius;
167 if (NDIM==2) variance=0.6*radius;
168 if (NDIM==3) variance=0.5*radius;
169 long maxrank=10*npoint_within_volume;
170 long rank=0;
171 for (int r=0; r<maxrank; ++r) {
172 auto tmp = gaussian_random_distribution(origin, variance);
173 if (not is_in_cell(tmp)) continue;
174 if (is_in_sphere(tmp)) ++rank;
175 grid.push_back(tmp);
176 if (rank==npoint_within_volume) break;
177 }
178 if (do_print) {
179 print("origin ",origin);
180 print("radius ",radius);
181 print("grid points in volume ",rank);
182 print("total grid points ",grid.size());
183 print("ratio ",rank/double(grid.size()));
184 print("volume element ",volume()/rank);
185 }
186 return grid;
187 }

190 return origin;
191 }

193 private:

195 double volume() const {
196 MADNESS_CHECK(NDIM>0 and NDIM<4);
197 if (NDIM==1) return 2.0*radius;
198 else if (NDIM==2) return constants::pi*radius*radius;
199 else if (NDIM==3) return 4.0 / 3.0 * constants::pi * std::pow(radius, 3.0);
200 else {
201 MADNESS_EXCEPTION("invalid NDIM",1);
202 return 0.0;
203 }

206 static Vector<double,NDIM> gaussian_random_distribution(const Vector<double,NDIM> origin, double variance) {
207 std::random_device rd{};
208 std::mt19937 gen{rd()};
209 Vector<double,NDIM> result;
210 for (size_t i = 0; i < NDIM; ++i) {
211 std::normal_distribution<> d{origin[i], variance};
212 result[i]=d(gen);
213 }
214
215 return result;
216 }

296 /// ctor takes centers of the grid and the grid parameters
297 molecular_grid(const std::vector<Vector<double,NDIM>> origins, const LowRankFunctionParameters& params)
298 : centers(origins)

300 if (centers.size()==0) centers.push_back(Vector<double,NDIM>(0) );
301 // treat harmonics and twostage like random sampling for constructing the molecular grid
302 if (params.gridtype()=="random" or params.gridtype()=="adaptive" or params.gridtype()=="harmonics" or params.gridtype()=="twostage") grid_builder=std::make_shared<randomgrid<NDIM>>(params.volume_element(),params.radius());
303 // else if (params.gridtype()=="cartesian") grid_builder=std::make_shared<cartesian_grid<NDIM>>(params.volume_element(),params.radius());
304 else if (params.gridtype()=="dftgrid") {
305 if constexpr (NDIM==3) {
306 grid_builder=std::make_shared<dftgrid<NDIM>>(params.volume_element(),params.radius());
307 } else {
308 MADNESS_EXCEPTION("no dft grid with NDIM != 3",1);
309 }
310 } else {
311 MADNESS_EXCEPTION("no such grid type",1);
312 }

316 /// ctor takes centers of the grid and the grid builder
317 molecular_grid(const std::vector<Vector<double,NDIM>> origins, std::shared_ptr<gridbase> grid)
318 : centers(origins), grid_builder(grid) {
319 if (centers.size()==0) centers.push_back({0,0,0});
320 }
321
322 /// ctor takes molecule and grid builder
323 molecular_grid(const Molecule& molecule, std::shared_ptr<gridbase> grid)
324 : molecular_grid(molecule.get_all_coords_vec(),grid) {}
325
326 std::vector<Vector<double,NDIM>> get_grid() const {
327 MADNESS_CHECK_THROW(grid_builder,"no grid builder given in molecular_grid");
328 MADNESS_CHECK_THROW(centers.size()>0,"no centers given in molecular_grid");
329
330 // A point generated around centers[ic] is kept iff centers[ic] is its
331 // nearest center — i.e. it lies in the Voronoi cell of its generator.
332 // This eliminates the over-density that arises when neighbouring
333 // per-center Gaussian clouds overlap. With a single center the test
334 // is vacuous.
335 const bool do_voronoi = (centers.size() > 1);
336 auto in_own_voronoi_cell = [&](const Vector<double,NDIM>& p, const std::size_t ic) {
337 double d_own = (p - centers[ic]).normf();
338 for (std::size_t jc = 0; jc < centers.size(); ++jc) {
339 if (jc == ic) continue;
340 if ((p - centers[jc]).normf() < d_own) return false;
341 }
342 return true;
343 };
344
345 std::vector<Vector<double,NDIM>> grid;
346 for (std::size_t ic = 0; ic < centers.size(); ++ic) {
347 const auto& coords = centers[ic];
348 print("atom sites",coords);
349 std::vector<Vector<double,NDIM>> atomgrid;
350 if (auto builder=dynamic_cast<dftgrid<NDIM>*>(grid_builder.get())) {
351 if constexpr (NDIM==3) {
352 dftgrid<NDIM> b1(builder->builder.get_nradial(),builder->builder.get_angular_order(),coords);
353 atomgrid=b1.get_grid();
354 } else {
355 MADNESS_EXCEPTION("no DFT grid for NDIM /= 3",1);
356 }
357 } else if (auto builder=dynamic_cast<randomgrid<NDIM>*>(grid_builder.get())) {
358 randomgrid<NDIM> b1(builder->get_volume_element(),builder->get_radius(),coords);
359 atomgrid=b1.get_grid();
360 } else {
361 MADNESS_EXCEPTION("no such grid builder",1);
362 }
363
364 if (do_voronoi) {
365 std::size_t kept = 0;
366 for (const auto& p : atomgrid) {
367 if (in_own_voronoi_cell(p, ic)) {
368 grid.push_back(p);
369 ++kept;
370 }
371 }
372 print("center",ic,"generated",atomgrid.size(),"kept",kept,"after Voronoi filter");
373 } else {
374 grid.insert(grid.end(),atomgrid.begin(),atomgrid.end());
375 }
376 }
377 return grid;
378 }

380 private:
381 std::vector<Vector<double,NDIM>> centers;
382 std::shared_ptr<gridbase> grid_builder;
383

551 typedef std::tuple<

570 resultT operator()(
571 const std::vector<Function<T,LDIM>>& rhs_batch,
572 const std::vector<Function<T,LDIM>>& a,
573 const std::vector<Function<T,LDIM>>& b,
574 const Tensor<T>& coeff,
575 const Tensor<double>& expnt,
576 const double lo,
577 const double thresh,
578 const double mu,
579 const bool p1_is_first) const {
580 World& world = rhs_batch.front().world();
581 // reconstruct operator from tensors so every task has identical numerics
582 auto f12_local = SeparatedConvolution<T,LDIM>(world, coeff, expnt, lo, thresh,
583 FunctionDefaults<LDIM>::get_bc().lattice_range(),
584 FunctionDefaults<LDIM>::get_k(), false, mu);
585
586 auto tmp2 = zero_functions_compressed<T,LDIM>(world, rhs_batch.size());
587 if (a.empty()) {
588 tmp2 = apply(world, f12_local, rhs_batch);
589 } else {
590 for (std::size_t ia = 0; ia < a.size(); ++ia) {
591 const auto& premultiply = p1_is_first ? a[ia] : b[ia];
592 const auto& postmultiply = p1_is_first ? b[ia] : a[ia];
593 auto tmp = copy(world, rhs_batch);
594 if (premultiply.is_initialized()) tmp = rhs_batch * premultiply;
595 auto tmp1 = apply(world, f12_local, tmp);
596 if (postmultiply.is_initialized()) tmp1 = tmp1 * postmultiply;
597 tmp2 += tmp1;
598 }
599 }
600 return truncate(tmp2,thresh*0.3);
601 }

606 // functor is now \sum_i a_i(1) b_i(2) f12
607 // result(1) = \sum_i \int a_i(1) f(1,2) b_i(2) rhs(2) d2
608 // = \sum_i a_i(1) \int f(1,2) b_i(2) rhs(2) d2

610 // MacroTask cannot serialize uninitialized Function objects. Replace any
611 // uninitialized entries (representing an implicit factor of 1) with an
612 // explicit constant-1 Function so the cloud transfer is well-defined.
613 auto sanitize = [&world](const std::vector<Function<T,LDIM>>& v) {
614 bool any_uninit=false;

616 if (not any_uninit) return v;
617 Function<T,LDIM> one = FunctionFactory<T,LDIM>(world)
618 .functor([](const Vector<double,LDIM>& r){return T(1);});
619 std::vector<Function<T,LDIM>> out=v;
620 for (auto& f : out) if (not f.is_initialized()) f=one;
621 return out;
622 };

627 return mt(rhs, a_safe, b_safe,
628 f12->stored_coeffs, f12->stored_exponents,
629 f12->info.lo, f12->info.thresh, f12->info.mu,
630 p1.is_first());

633 typename Tensor<T>::scalar_type norm2() const override {
634 const Function<T, LDIM> one = FunctionFactory<T, LDIM>(world()).f(
635 [](const Vector<double, LDIM>& r) { return 1.0; });
636 std::vector<Function<T, LDIM>> pre, post;

643 post = (b.front().is_initialized()) ? b : std::vector<Function<T, LDIM>>(sz, one);

646 const SeparatedConvolution<T,LDIM>& f12a=*(f12);

648 // the generic outer-product construction for SepConvs built from
649 // raw (c, α) tensors whose OperatorInfo type is UNDEFINED.
650 auto build_f12sq = [&]() -> SeparatedConvolution<T,LDIM> {
651 if (SeparatedConvolution<T,LDIM>::can_combine(f12a, f12a)) {
652 return SeparatedConvolution<T,LDIM>::combine(f12a, f12a);

656 "stored (c, α) available for generic fallback");

658 world(), f12a.stored_coeffs, f12a.stored_exponents,
659 f12a.stored_coeffs, f12a.stored_exponents,
660 f12a.info.lo, f12a.info.thresh, /*prune=*/true);
661 }
662 };
663 const SeparatedConvolution<T,LDIM> f12sq = build_f12sq();

665 // \int f(1,2)^2 d1d2 = \int f(1,2)^2 pre(1)^2 post(2)^2 d1 d2
666 // || \sum_i f(1,2) a_i(1) b_i(2) || = \int ( \sum_{ij} a_i(1) a_j(1) f(1,2)^2 b_i(1) b_j(2) ) d1d2
667 std::vector<Function<T,LDIM>> aa,bb;
668 for (std::size_t i=0; i<pre.size(); ++i) {
669 aa=append(aa,(pre[i]*pre));
670 bb=append(bb,(post[i]*post));
671 }
672// typename Tensor<T>::scalar_type term1 =madness::inner(mul(world(),post,post),f12sq(mul(world(),pre,pre)));
673 typename Tensor<T>::scalar_type term1 =madness::inner(bb,f12sq(aa));
674 return sqrt(term1);

676 }

678 T operator()(const Vector<double,NDIM>& r) const override {
679
680 if (a.size()==0) return 0.0;
681 auto split = [](const Vector<double,NDIM>& r) {
682 Vector<double,LDIM> first, second;
683 for (size_t i=0; i<LDIM; ++i) {
684 first[i]=r[i];
685 second[i]=r[i+LDIM];
686 }
687 return std::make_pair(first,second);
688 };
689
690 double gamma=f12->info.mu;
691 auto [first,second]=split(r);
692
693
694 double result=0.0;
695 for (std::size_t ia=0; ia<a.size(); ++ia) {
696 // a[ia](first) reconstructs a non-reconstructed function, which fences: fine from the
697 // main thread, a deadlock from inside a task (e.g. projecting this functor), so
698 // reconstruct a and b before using the functor in FunctionFactory::functor()
699 double result1=1.0;
700 if (a[ia].is_initialized()) result1*=a[ia](first);
701 if (b[ia].is_initialized()) result1*=b[ia](second);
702 if (f12->info.type==OT_SLATER) result1*=exp(-gamma*(first-second).normf());
703 else if (f12->info.type==OT_F12) result1*=(1.0-exp(-gamma* madness::inner(first-second,first-second)));
704 else if (f12->info.type==OT_GAUSS) result1*=exp(-gamma* madness::inner(first-second,first-second));
705 else {
706 MADNESS_EXCEPTION("no such operator_type",1);
707 }
708 result+=result1;
709 }
710 return result;

961 /// remove linear dependencies in g and h; result stays in canonical form
962 /// f(1,2) = \sum_i g_i(1) h_i(2)
963 /// {g} and {h} are not necessarily orthogonal, but they are linearly independent
964 void remove_linear_dependencies(double tol=-1.0) {
965 if (tol<0.0) tol=rank_revealing_tol;
966 if (g.size()==0) return;
967 World& world=g.front().world();
968
969 Tensor<T> ovlp_g=matrix_inner(world,g,g);
970 Tensor<T> ovlp_h=matrix_inner(world,h,h);
971 auto [pg,Xg,Xg_inv]=rr_cholesky_matrix_and_reorder(g,ovlp_g,tol);
972 auto [ph,Xh,Xh_inv]=rr_cholesky_matrix_and_reorder(h,ovlp_h,tol);
973 // cite from rr_cholesky_matrix_and_reorder
974 // pv: reordered original, linearly-independent functions,
975 // X: the (r,r) transformation matrix from pv to orthonormalized functions:
976 // v_ortho_i = v_p X_ip = pv_j X_ji
977 // X_inv: its (r,p) "inverse" from original functions to orthonormalized functions:
978 // v_p = v_ortho_i X_inv_ip
979 // note that X is not the inverse of X_inv, that would be X_full
980 // g h = pg X X^(-1) X^(-T) X^T ph
981 // absorb the coupling matrix M = Xg . Xh^T into g to keep the canonical form
982 auto XXinv_g=inner(Xg,Xg_inv);
983 auto XXinv_h=inner(Xh,Xh_inv);
984 auto M = inner(XXinv_g,XXinv_h,1,1); // Xg . Xh^T
985 g=transform(world,pg,M);
986 h=ph;
987 }

994 /// perform RRCD on the overlap matrix of the functions, reorder the functions according to the pivoting,
995 /// and return the upper triangular matrix from the Cholesky decomposition
996
997 /// note: the rr_cholesky is more efficient than a full SVD for removing linear dependencies, but it does not
998 /// give an orthonormal basis, so an additional orthonormalization step is needed if orthogonality is desired.
999 /// To orthonormalize:
1000 /// auto [pv, X] = rr_cholesky_matrix_and_reorder(v,ovlp,tol);
1001 /// v_ortho=transform(world,pv,X); // v_ortho_i = sum_j v_j U_{ji}^(-1)
1002 /// @param[in] v the input functions
1003 /// @param[in] ovlp the overlap matrix of the input functions, will be destroyed in-place by the rr_cholesky
1004 /// @param[in] tol the tolerance for linear dependence in the rr_cholesky
1005 /// @return a tuple containing
1006 /// pv: reordered original, linearly-independent functions,
1007 /// X: the (r,r) transformation matrix from pv to orthonormalized functions:
1008 /// v_ortho_i = v_p X_pi = pv_j X_ji
1009 /// X_inv: its (r,p) "inverse" from original functions to orthonormalized functions:
1010 /// v_p = v_ortho_i X_inv_ip
1011 /// note that X is not the inverse of X_inv, that would be X_full
1012 /// but v_p(x) = pv_i(x) X_ij X_inv_jp is numerically the same
1013 static std::tuple<std::vector<Function<T,LDIM>>, Tensor<T>, Tensor<T>>
1014 rr_cholesky_matrix_and_reorder(const std::vector<Function<T,LDIM>>& v, Tensor<T> ovlp, const double tol) {
1015 int rank;
1016 Tensor<int> piv;
1017
1018 rr_cholesky(ovlp,tol,piv,rank); // destroys ovlp and gives back Upper ∆ Matrix from CCD
1019
1020 // rearrange and truncate the functions according to the pivoting of the rr_cholesky
1021 std::vector<Function<T,LDIM> > pv(rank);
1022 for(integer i=0;i<rank;++i) pv[i]=v[piv[i]];
1023
1024 // compute transformation matrix v -> w = v_ortho
1025 // w_i = v_p X_pi
1026 // = v_piv(i) X_ij
1027 // v_p = w_i X^(-1)_ip = w_i Xinv_ip
1028 // pv_i = piv(v_p) // pv is not orthogonal or normalized, just linearly independent
1029 // this returns: pv, X_ij, X_inv_ip
1030
1031 // no need to invert all of ovlp, only the upper left rank x rank block
1032 Tensor<T> U=ovlp(Slice(0,rank-1),Slice(0,rank-1));
1033 Tensor<T> Xinv=ovlp(Slice(0,rank-1),_); // has dimensions (rank,original_size)
1034 Tensor<T> X=inverse(U); // has dimensions (rank,rank)
1035
1036 // sanity check
1037 // result = |i~> = sum_r |i> U_{ir}^(-1)
1038 // with <i~|j~> = sum_{r} U_{ir}^(-1) <i|j> U_{jp}^(-1)
1039 // = sum_{r} U_{ir}^(-1) (U^T U)_{ij} U_{jr}^(-1)
1040 // = sum_{r} U_{ir}^(-1) U_{ip} U_{jp} U_{jp}^(-1)
1041 // = delta_{ij} // has matrix dimension (r,r)
1042
1043 // permute X_inv, so that the following identity holds:
1044 // also the same: v_p(x) = pv_i(x) X_ij X_inv_jq P_qp
1045 Tensor<T> tmp=copy(Xinv);
1046 for (int i=0; i<tmp.dim(1); ++i) Xinv(_,piv(i))=tmp(_,i);
1047
1048 return std::make_tuple(pv,X,Xinv);
1049 }

1082 /// compute the l2 error |functor - \sum_i g_ih_i|_2
1083
1084 /// \int (f(1,2) - gh(1,2))^2 = \int f(1,2)^2 - 2\int f(1,2) gh(1,2) + \int gh(1,2)^2
1085 /// since we are subtracting large numbers the numerics are sensitive, and NaN may be returned..
1086 double l2error(const LRFunctorBase<T,NDIM>& lrfunctor1) const {
1087
1088 timer t(world());
1089 t.do_print=do_print;
1090
1091 // \int f(1,2)^2 d1d2
1092 double term1 = lrfunctor1.norm2();
1093 term1=term1*term1;
1094 t.tag("computing term1");
1095
1096 // \int f(1,2) \sum_i g_i(1) h_i(2) d1d2
1097 // = \sum_i \int (\int f(1,2) g_i(1) d1) h_i(2) d2
1098 double term2=madness::inner(g,inner(lrfunctor1,h,p2,p1));
1099 t.tag("computing term2");
1100
1101 auto tmp1=matrix_inner(world(),h,h);
1102 auto tmp2=matrix_inner(world(),g,g);
1103 double term3=tmp1.trace(tmp2);
1104 t.tag("computing term3");
1105
1106 double arg=term1-2.0*term2+term3;
1107 if (arg<0.0) {
1108 print("negative l2 error");
1109 arg*=-1.0;
1110// throw std::runtime_error("negative argument in l2error");
1111 }
1112 double error=sqrt(arg)/sqrt(term1);
1113 if (world().rank()==0 and do_print) {
1114 print("term1,2,3, error",term1, term2, term3, " --",error);
1115 }

1120 /// Cheap relative L2 error via Pythagoras (canonical form only).
1121 ///
1122 /// When h is orthonormal AND g is the reprojection of h
1123 /// (g_k = <f(r1,r2) | h_k(r2)>_{r2}), the following identity holds:
1124 /// ||f - sum_k g_k h_k||^2 = ||f||^2 - sum_k ||g_k||^2
1125 /// (cross and self terms collapse to the same sum_k ||g_k||^2).
1126 /// No functor applications are needed — only norms of the stored g.
1127 ///
1128 /// @param[in] f_norm_sq precomputed ||f||^2 (typically lrfunctor.norm2()^2)
1129 /// @pre the LRF is in canonical form and g holds the reprojected factors.
1130 double l2error_pythagoras(double f_norm_sq) const {
1131 MADNESS_CHECK_THROW(f_norm_sq > 0.0,
1132 "l2error_pythagoras requires a positive ||f||^2");
1133 if (g.size()==0) return 0.0;
1134 auto gnorms = norm2s(g.front().world(), g);
1135 double sum_gk_sq = 0.0;
1136 for (double n : gnorms) sum_gk_sq += n * n;
1137 double arg = f_norm_sq - sum_gk_sq;
1138 if (arg < 0.0) arg = -arg; // numerical noise near convergence
1139 // print("l2 error via Pythagoras: (abs., rel.)", sqrt(arg),sqrt(arg) / sqrt(f_norm_sq));
1140 return sqrt(arg) / sqrt(f_norm_sq);
1141 }

1265 /// Factory class to compute a low-rank approximation of a given hi-dimensional function
1266 /// f(r1,r2) using randomized projection and rank-revealing Cholesky decomposition (RRCD).
1267 ///
1268 /// The approximation takes the canonical form
1269 /// f(r1,r2) ≈ Σ_i g_i(r1) h_i(r2)
1270 /// where h are orthonormalized backprojected functions and g their reprojections.
1271 ///
1272 /// ## Error analysis
1273 ///
1274 /// The L2 error ||f - f_approx|| has three independent sources that add in quadrature:
1275 ///
1276 /// ε_total² ≈ ε_sampling² + ε_truncation² + ε_conditioning²
1277 ///
1278 /// ### 1. Sampling error (grid coverage)
1279 ///
1280 /// The grid probes f's column space via localized Gaussians. If the grid misses
1281 /// regions where f has significant weight, those contributions cannot be represented.
1282 /// Controlled by: volume_element (ve), radius, center placement.
1283 ///
1284 /// Heuristic: for well-localized functions (Gaussians on atomic centers), ve ≈ 0.1
1285 /// with radius ≈ 2-3 suffices for errors down to ~1e-3. Finer grids increase
1286 /// Y-formation cost linearly (O(grid_size) functor applications) but the rank
1287 /// saturates quickly — most extra grid points are linearly dependent.
1288 ///
1289 /// ### 2. Truncation error (rank reduction)
1290 ///
1291 /// The RRCD with tolerance tol discards Y components with overlap eigenvalues < tol.
1292 /// For Gaussian kernels the eigenvalues decay rapidly, giving truncation error that
1293 /// scales roughly as tol^(0.3-0.5) (empirically). Setting tol = ε² gives truncation
1294 /// error ~ ε.
1295 ///
1296 /// ### 3. Conditioning (eliminated by power-iteration algorithm)
1297 ///
1298 /// The current algorithm orthonormalizes h = f·Y (not Y), so the overlap
1299 /// <h|h> has eigenvalues weighted by σ_k² (singular values of f). Rank
1300 /// reduction on <h|h> is always well-conditioned — no half-metric or
1301 /// full-metric amplification. The result is canonical by construction
1302 /// (no coupling metric), and conditioning is not an error source.
1303 ///
1304 /// ### Parameter heuristics for a target error ε
1305 ///
1306 /// tol = ε² (truncation error ~ √tol ~ ε)
1307 /// ve ~ C · ε^(2/LDIM) with C = 2/1/0.5 for LDIM=1/2/3
1308 ///
1309 /// ### Algorithm (project())
1310 ///
1311 /// 1. Build grid around origins (Gaussian-distributed, σ² = 0.5·radius)
1312 /// 2. Form Y basis: multi-scale Gaussian RHS at grid points
1313 /// 3. Power-iteration projection: Y → h_raw → rr_cholesky(<h|h>) → h_ortho → g
1314 /// 4. Tol sweep: incrementally tighten rr_cholesky tolerance, reuse cached <h|h>
1315 /// 5. If grid-limited: halve ve, rebuild Y, re-project from scratch
1316 /// 6. If thresh-limited: tighten MRA thresh, rebuild Y, re-project
1317 /// 7. L2 error via Pythagoras: ||f||² − Σ||g_k||² (no functor applications)
1318 ///
1319 /// The project() method automates parameter selection from a single
1320 /// target L2 error.
1321 template<typename T, std::size_t NDIM, std::size_t LDIM=NDIM/2>

1414 /// Numerically stable projection using power-iteration.
1415 ///
1416 /// this method orthonormalizes h = f·Y. The overlap <h|h> has eigenvalues
1417 /// weighted by σ_k² (singular values of f), making the rank reduction well-conditioned.
1418 /// The result is canonical by construction (no metric).
1419 ///
1420 /// Algorithm:
1421 /// 1. rr_cholesky(<Y|Y>, tol_coarse) → pY (coarse rank reduction)
1422 /// 2. h_raw = inner(f, pY, p1, p1) (backproject to r2)
1423 /// 3. rr_cholesky(<h_raw|h_raw>, tol) → ph (well-conditioned rank)
1424 /// 4. orthonormalize ph → h_ortho (via Cholesky factor)
1425 /// 5. g = inner(f, h_ortho, p2, p1) (reproject to r1)
1426 /// 6. return LowRankFunction(g, h_ortho) (canonical)
1427 ///
1428 /// @param[in] lrfunctor the functor
1429 /// @param[in] Y the probe basis from Yformer
1430 /// @param[in] tol the target Cholesky tolerance
1431 /// @return canonical LowRankFunction
1432 LowRankFunction<T,NDIM> project_from_Y_stable(
1433 const LRFunctorBase<T,NDIM>& lrfunctor,
1434 const std::vector<Function<T,LDIM>>& Y,
1435 const double tol) const
1436 {
1437 World& world = lrfunctor.world();
1438 timer t1(world);
1439 t1.do_print = true;
1440
1441 double tight_thresh = FunctionDefaults<LDIM>::get_thresh() * 0.1;
1442
1443 // Step 1: coarse rank reduction on Y (cheap, removes exact lindep only)
1444 // Use TIGHT tolerance so we don't prematurely discard info; the
1445 // authoritative rank reduction happens on <h|h> below.
1446 double tol_coarse = std::max(1.e-14, tol * 0.01);
1447 auto ovlp_Y = matrix_inner(world, Y, Y);
1448 auto [pY, Ut_Y, Xinv_Y] = LowRankFunction<T,NDIM>::rr_cholesky_matrix_and_reorder(
1449 Y, ovlp_Y, tol_coarse);
1450 t1.tag("stable: rr_cholesky on Y (coarse)");
1451 print("stable: Y.size =", Y.size(), "pY.size =", pY.size());
1452
1453 // Step 2: backproject pY to r2 space — h_raw inherits σ_k² conditioning
1454 auto h_raw = truncate(inner(lrfunctor, pY, p1, p1), tight_thresh);
1455 t1.tag("stable: backprojection (h_raw)");
1456
1457 // Step 3: rank reduction on <h|h> — well-conditioned
1458 auto ovlp_h = matrix_inner(world, h_raw, h_raw);
1459 auto [ph, Ut_h, Xinv_h] = LowRankFunction<T,NDIM>::rr_cholesky_matrix_and_reorder(
1460 h_raw, ovlp_h, tol);
1461 t1.tag("stable: rr_cholesky on h");
1462 print("stable: h_raw.size =", h_raw.size(), "ph.size =", ph.size());
1463
1464 // Step 4: orthonormalize ph → h_ortho via Cholesky factor X = U^{-1}
1465 // Since <h|h> is well-conditioned, this orthonormalization is stable
1466 auto h_ortho = truncate(transform(world, ph, Ut_h), tight_thresh);
1467 t1.tag("stable: orthonormalization");
1468
1469 // Step 5: reproject to r1 space — g lives in r1 and is well-behaved
1470 auto g = truncate(inner(lrfunctor, h_ortho, p2, p1), tight_thresh);
1471 t1.tag("stable: reprojection (g)");
1472
1473 // Step 6: canonical form, no metric
1474 LowRankFunction<T,NDIM> result(g, h_ortho, tol);
1475 print("stable: final rank =", g.size());
1476
1477 return result;
1478 }

1540 /// Stable projection using power iteration.
1541 ///
1542 /// If prev state is empty, does full projection (Y → h → ortho → g).
1543 /// If prev has content, incrementally extends: rr_cholesky on ovlp_h with
1544 /// tighter tol, orthonormalize and reproject only the new functions.
1545 ///
1546 /// The overlap matrix ovlp_h is cached and reused across calls for pivot
1547 /// consistency (parallel matrix_inner is non-deterministic).
1548 ProjectionState project_stable(
1549 const LRFunctorBase<T,NDIM>& lrfunctor,
1550 const std::vector<Function<T,LDIM>>& Y,
1551 const double tol,
1552 const ProjectionState& prev) const
1553 {
1554 World& world = lrfunctor.world();
1555 timer t1(world);
1556 t1.do_print = true;
1557 double tight_thresh = FunctionDefaults<LDIM>::get_thresh() * 0.1;
1558
1559 ProjectionState state;
1560 state.tol = tol;
1561 // cache ||f||^2 for cheap Pythagoras l2error (canonical form)
1562 if (prev.f_norm_sq > 0.0) {
1563 state.f_norm_sq = prev.f_norm_sq;
1564 } else {
1565 double fn = lrfunctor.norm2();
1566 state.f_norm_sq = fn * fn;
1567 t1.tag("stable: ||f||^2");
1568 }
1569
1570 if (prev.rank() == 0) {
1571 // --- full projection ---
1572 // coarse rank reduction on Y
1573 double tol_coarse = std::max(1.e-14, tol * 0.01);
1574 auto ovlp_Y = matrix_inner(world, Y, Y);
1575 auto [pY, Ut_Y, Xinv_Y] = LowRankFunction<T,NDIM>::rr_cholesky_matrix_and_reorder(
1576 Y, ovlp_Y, tol_coarse);
1577 t1.tag("stable: rr_cholesky on Y");
1578
1579 // backproject to h_raw
1580 state.h_raw = truncate(inner(lrfunctor, pY, p1, p1), tight_thresh);
1581 t1.tag("stable: backprojection (h_raw)");
1582
1583 // cache <h|h> and do authoritative rank reduction
1584 state.ovlp_h = matrix_inner(world, state.h_raw, state.h_raw);
1585 auto [ph, Ut_h, Xinv_h] = LowRankFunction<T,NDIM>::rr_cholesky_matrix_and_reorder(
1586 state.h_raw, copy(state.ovlp_h), tol);
1587 t1.tag("stable: rr_cholesky on h");
1588
1589 state.h_ortho = truncate(transform(world, ph, Ut_h), tight_thresh);
1590 t1.tag("stable: orthonormalization");
1591
1592 state.g = truncate(inner(lrfunctor, state.h_ortho, p2, p1), tight_thresh);
1593 t1.tag("stable: reprojection (g)");
1594 LowRankFunction<T,NDIM> lrf(state.g, state.h_ortho, tol);
1595 double current_error = lrf.l2error_pythagoras(state.f_norm_sq);
1596 t1.tag("stable: compute error in optimization");
1597
1598
1599 print("stable: pY =", pY.size(), "h_raw =", state.h_raw.size(),
1600 "rank =", state.g.size());
1601
1602 } else {
1603 // --- incremental tol-tightening on <h|h> ---
1604 // reuse prev.h_raw and prev.ovlp_h (must be same — cached)
1605 state.h_raw = prev.h_raw;
1606 state.ovlp_h = prev.ovlp_h;
1607
1608 Tensor<T> ovlp_work = copy(state.ovlp_h);
1609 auto [ph, Ut_h, Xinv_h] = LowRankFunction<T,NDIM>::rr_cholesky_matrix_and_reorder(
1610 state.h_raw, ovlp_work, tol);
1611 t1.tag("stable: rr_cholesky on h (incremental)");
1612
1613 long r_prev = prev.rank();
1614 long r_new = ph.size();
1615
1616 if (r_new <= r_prev) {
1617 state.h_ortho = prev.h_ortho;
1618 state.g = prev.g;
1619 } else {
1620 // only orthonormalize and reproject the new functions
1621 Tensor<T> X_new_cols = Ut_h(_, Slice(r_prev, r_new - 1));
1622 auto h_ortho_new = truncate(transform(world, ph, X_new_cols), tight_thresh);
1623 t1.tag("stable: orthonormalization (new only)");
1624 auto g_new = truncate(inner(lrfunctor, h_ortho_new, p2, p1), tight_thresh);
1625 t1.tag("stable: reprojection (new only)");
1626
1627 state.h_ortho = append(prev.h_ortho, h_ortho_new);
1628 state.g = append(prev.g, g_new);
1629 }
1630 print("stable incremental: r_prev =", r_prev, "r_new =", state.g.size());
1631 }
1632
1633 return state;
1634 }

1636 /// Diagnose the dominant error source.
1637 /// With the stable (canonical) algorithm, conditioning is not an issue.
1638 /// Thresh-limited only if:
1639 /// - the last tol-tightening did not improve the error by at least 20%
1640 /// (error >= 0.8 * prev_error), AND
1641 /// - sqrt(rank) * thresh > 0.3 * error (realistic noise combination), AND
1642 /// - tol is at the floor (~1e-14), i.e. tol-tightening has no more room.
1643 /// The sqrt(rank) factor replaces the worst-case rank factor: for canonical
1644 /// form with uncorrelated truncation noise, combined error scales like
1645 /// sqrt(rank) * thresh, not rank * thresh. The rank-linear bound triggers
1646 /// spuriously at moderate rank. The tol-floor guard prevents switching to
1647 /// expensive thresh work when grid augmentation hasn't been tried yet.
1648 /// Tol-limited: tol probe on ovlp_h shows rank increase (preferred when available).
1649 /// Grid-limited: neither of the above — try more grid points.
1650 std::string diagnose_error(double error, double prev_error, long rank,
1651 double tol, double thresh,
1652 const Tensor<T>& ovlp_h) const {
1653 double eps_trunc = std::sqrt(double(rank)) * thresh;
1654 bool error_stalled = (prev_error > 0 && error >= prev_error * 0.8);
1655 bool tol_at_floor = (tol <= 1.01e-14);
1656 print(" diagnosis: sqrt(rank)*thresh =", eps_trunc, "error =", error,
1657 "rank =", rank, "tol =", tol,
1658 "stalled =", error_stalled, "tol_at_floor =", tol_at_floor);
1659
1660 // tol probe on <h|h> — check first, prefer tol if available
1661 double tol_probe = std::max(1.e-14, tol * 0.01);
1662 if (tol_probe < tol && ovlp_h.size() > 0) {
1663 Tensor<T> ovlp_copy = copy(ovlp_h);
1664 int rank_probe;
1665 Tensor<integer> piv_probe;
1666 rr_cholesky(ovlp_copy, tol_probe, piv_probe, rank_probe);
1667 print(" tol probe on <h|h>: rank", rank, "->", rank_probe);
1668 if (rank_probe > rank * 1.1) {
1669 print(" -> tol-limited");
1670 return "tol";
1671 }
1672 }
1673
1674 // thresh: only when error stalled AND realistic noise bound is significant
1675 // AND we've already exhausted tol headroom (else grid should be tried first).
1676 if (error_stalled && tol_at_floor && eps_trunc > 0.3 * error) {
1677 print(" -> thresh-limited");
1678 return "thresh";
1679 }
1680 print(" -> grid-limited");
1681 return "grid";
1682 }

1684 /// Adaptively project a high-dimensional functor to a low-rank representation.
1685 ///
1686 /// Uses the numerically stable power-iteration algorithm:
1687 /// Y → h = f·Y → orthonormalize h → g = f·h_ortho → canonical LRF
1688 ///
1689 /// Single refinement loop with three actions:
1690 /// - tol-limited: tighten tol on <h|h>, incrementally extend h_ortho and g
1691 /// - grid-limited: halve ve and re-project from scratch (denser Y sampling)
1692 /// - thresh-limited: tighten MRA thresh, reform everything
1693 /// - tol-degraded: revert to best state, route to thresh or grid
1694 ///
1695 /// @param[in] lrfunctor the high-dimensional functor to approximate
1696 /// @param[in] target_l2error the desired L2 error (default: FunctionDefaults thresh)
1697 /// @param[in] max_iter maximum refinement iterations (default: 5)
1698 /// @return canonical low-rank approximation
1699 LowRankFunction<T,NDIM> project(
1700 const LRFunctorBase<T,NDIM>& lrfunctor,
1701 const double target_l2error = FunctionDefaults<NDIM>::get_thresh(),
1702 const int max_iter = 5) const
1703 {
1704 World& world = lrfunctor.world();
1705 timer t1(world);
1706 t1.do_print = true;
1707 const double eps = target_l2error;
1708 const long max_Y_size = 5000;
1709
1710 double tol = std::max(1.e-14, std::min(1.e-3, eps * eps));
1711 double current_thresh = FunctionDefaults<LDIM>::get_thresh();
1712 double ve = parameters.volume_element();
1713
1714 if (world.rank()==0) print("project: eps =", eps, "tol =", tol,
1715 "ve =", ve, "thresh =", current_thresh);
1716 t1.tag("project: parameter setup");
1717
1718 LowRankFunctionParameters local_params = parameters;
1719 local_params.set_derived_value("volume_element", ve);
1720 local_params.set_derived_value("tol", tol);
1721
1722 std::vector<Vector<double,LDIM>> grid;
1723 if (world.rank()==0) grid=build_grid(ve);
1724 // broadcast grid from rank0
1725 world.gop.broadcast_serializable(grid, 0);
1726 if (world.rank()==0) print("project: initial grid size", grid.size());
1727 t1.tag("project: grid construction");
1728
1729 auto Y = form_Y(lrfunctor, grid, local_params);
1730 MADNESS_CHECK_THROW(!Y.empty(), "project: no basis functions");
1731 if (world.rank()==0) print("project: initial Y size", Y.size());
1732 t1.tag("project: Yforming");
1733
1734 // initial stable projection
1735 ProjectionState empty_state;
1736 auto state = project_stable(lrfunctor, Y, tol, empty_state);
1737 auto lrf = LowRankFunction<T,NDIM>(state.g, state.h_ortho,
1738 tol);
1739 t1.tag("project: initial projection");
1740
1741 double current_error = lrf.l2error_pythagoras(state.f_norm_sq);
1742 long current_rank = state.rank();
1743 if (world.rank()==0) print("project: iter 0, l2error =", current_error,
1744 "rank =", current_rank, "target =", eps);
1745 t1.tag("project: l2error");
1746
1747 if (current_error <= eps) {
1748 if (world.rank()==0) print("project: converged at iteration 0");
1749 return lrf;
1750 }
1751
1752 // ThreshGuard for optional thresh-tightening
1753 std::unique_ptr<ThreshGuard> thresh_guard;
1754 double prev_error = -1.0;
1755 int iter = 0;
1756
1757 // track best state seen across all iterations
1758 auto best_lrf = lrf;
1759 double best_error = current_error;
1760
1761 while (iter <= max_iter) {
1762 if (state.ovlp_h.size() == 0) state.ovlp_h=matrix_inner(world,state.h_ortho,state.h_ortho);
1763 auto diagnosis = diagnose_error(current_error, prev_error, current_rank,
1764 tol, current_thresh, state.ovlp_h);
1765 prev_error = current_error;
1766
1767 // --- tol-limited: try incremental tol-tightening ---
1768 if (diagnosis == "tol") {
1769 auto prev_state = state;
1770 auto prev_lrf = lrf;
1771 double prev_tol = tol;
1772
1773 tol = std::max(1.e-14, tol * 0.01);
1774 state = project_stable(lrfunctor, Y, tol, prev_state);
1775 lrf = LowRankFunction<T,NDIM>(state.g, state.h_ortho,
1776 tol);
1777 t1.tag("project: tol refinement iter " + std::to_string(iter));
1778
1779 current_error = lrf.l2error_pythagoras(state.f_norm_sq);
1780 current_rank = state.rank();
1781 print("project: iter", iter, "(tol), l2error =",
1782 current_error, "rank =", current_rank);
1783 t1.tag("project: l2error iter " + std::to_string(iter));
1784 if (current_error <= eps) return lrf;
1785 if (current_error < best_error) {
1786 best_error = current_error;
1787 best_lrf = lrf;
1788 }
1789
1790 if (current_error > prev_error * 1.1) {
1791 // tol made it worse — revert. Tol-degradation is a strong
1792 // thresh-limit signal: we're trying to resolve singular values
1793 // smaller than the current MRA thresh can represent, so the
1794 // new directions are numerical noise. Route to thresh-tightening
1795 // (if not already done); otherwise grid augmentation is the
1796 // only remaining avenue.
1797 print("project: tol degraded, reverting");
1798 state = prev_state;
1799 lrf = prev_lrf;
1800 tol = prev_tol;
1801 current_error = prev_error;
1802 current_rank = state.rank();
1803 diagnosis = thresh_guard ? "grid" : "thresh";
1804 // NB: do NOT ++iter here — the branch below does so
1805 } else {
1806 continue; // tol succeeded — don't count as iteration
1807 }
1808 }
1809
1810 // --- grid-limited (or tol-degraded fallthrough): fresh Y at finer ve ---
1811 if (diagnosis == "grid") {
1812 ++iter;
1813 if (iter>max_iter) break;
1814 ve *= 0.5;
1815 print("project: grid-limited, halving ve to", ve, "and re-projecting from scratch");
1816 local_params.set_derived_value("volume_element", ve);
1817
1818 auto new_grid = build_grid(ve, max_Y_size);
1819 if (new_grid.empty()) break;
1820 print("project: new grid size", new_grid.size());
1821
1822 Y = form_Y(lrfunctor, new_grid, local_params);
1823 t1.tag("project: grid Yforming iter " + std::to_string(iter));
1824 if (Y.empty()) break;
1825 print("project: new Y size", Y.size());
1826
1827 // reset tol and prev_error for a fresh tol sweep
1828 tol = std::max(1.e-14, std::min(1.e-3, eps * eps));
1829 prev_error = -1.0;
1830
1831 ProjectionState fresh;
1832 state = project_stable(lrfunctor, Y, tol, fresh);
1833 lrf = LowRankFunction<T,NDIM>(state.g, state.h_ortho,
1834 tol);
1835 t1.tag("project: grid projection iter " + std::to_string(iter));
1836
1837 current_error = lrf.l2error_pythagoras(state.f_norm_sq);
1838 current_rank = state.rank();
1839 print("project: iter", iter, "(grid ve=", ve, "), l2error =",
1840 current_error, "rank =", current_rank);
1841 t1.tag("project: l2error iter " + std::to_string(iter));
1842 if (current_error <= eps) return lrf;
1843
1844 // stagnation: grid halving didn't beat the previous best
1845 if (current_error >= best_error * 0.9) {
1846 print("project: grid halving stagnated (best=", best_error,
1847 "), stopping");
1848 break;
1849 }
1850 if (current_error < best_error) {
1851 best_error = current_error;
1852 best_lrf = lrf;
1853 }
1854 continue;
1855 }
1856
1857 // --- thresh-limited: tighten thresh, reform everything ---
1858 if (diagnosis == "thresh") {
1859 ++iter;
1860 if (iter>max_iter) break;
1861 if (thresh_guard) {
1862 print("project: thresh already tightened, giving up");
1863 break;
1864 }
1865 current_thresh *= 0.1;
1866 if (world.rank() == 0) print("project: tightening thresh to", current_thresh);
1867 thresh_guard = std::make_unique<ThreshGuard>(current_thresh,
1868 const_cast<LRFunctorBase<T,NDIM>&>(lrfunctor));
1869 t1.tag("project: thresh tightening");
1870
1871 Y = form_Y(lrfunctor, grid, local_params);
1872 if (Y.empty()) break;
1873 t1.tag("project: tight Yforming");
1874
1875 tol = std::max(1.e-14, std::min(1.e-3, eps * eps));
1876 ProjectionState empty;
1877 state = project_stable(lrfunctor, Y, tol, empty);
1878 lrf = LowRankFunction<T,NDIM>(state.g, state.h_ortho,
1879 tol);
1880 t1.tag("project: tight projection");
1881
1882 current_error = lrf.l2error_pythagoras(state.f_norm_sq);
1883 current_rank = state.rank();
1884 if (world.rank()==0) print("project: iter", iter, "(thresh), l2error =",
1885 current_error, "rank =", current_rank);
1886 t1.tag("project: l2error iter " + std::to_string(iter));
1887 if (current_error <= eps) return lrf;
1888 }
1889 }
1890
1891 if (world.rank()==0) print("WARNING project: did NOT converge to target", eps, "; achieved", best_error);
1892 return best_lrf;
1893 }

1896 /// Build a Taylor-expansion-based Y basis spanning the range of an F12 operator.
1897 ///
1898 /// For each Gaussian term c_m exp(-alpha_m r^2) in the operator's GFit
1899 /// expansion and each multi-index k with |k| <= max_order, emits
1900 /// phi_{m,k}(r) = r^k * exp(-alpha_m |r|^2)
1901 /// when the Taylor coefficient |c_m * prod_d (2 alpha_m)^{k_d}/k_d!|
1902 /// exceeds coeff_cutoff. These monomial-Gaussians probe the operator's
1903 /// range analytically rather than by random sampling.
1904 std::vector<Function<T,LDIM>> build_taylor_y(
1905 const LRFunctorF12<T,NDIM>& functor,
1906 const int max_order,
1907 const double coeff_cutoff) const
1908 {
1909 World& world = functor.world();
1910 auto f12 = functor.get_f12();
1911 // Prefer the SepConv's cached (c, α) — populated by both ctors.
1912 // Falling back to GFit(info) requires info.hi > 0, which the
1913 // explicit (coeff, expnt, lo, thresh) ctor leaves at zero.
1914 Tensor<T> coeffs = f12->stored_coeffs;
1915 Tensor<double> expnts = f12->stored_exponents;
1916 if (coeffs.size() == 0) {
1917 GFit<T, LDIM> fit(f12->info);
1918 coeffs = fit.coeffs();
1919 expnts = fit.exponents();
1920 }
1921 const long M = coeffs.dim(0);
1922
1923 auto multi_indices = generate_multi_indices(max_order);
1924
1925 std::vector<Function<T,LDIM>> Y;
1926 long skipped = 0;
1927 for (long m = 0; m < M; ++m) {
1928 double cm = coeffs(m);
1929 double am = expnts(m);
1930 for (const auto& kidx : multi_indices) {
1931 double coeff_mk = cm;
1932 for (std::size_t d = 0; d < LDIM; ++d) {
1933 for (int j = 0; j < kidx[d]; ++j) {
1934 coeff_mk *= (2.0 * am) / double(j + 1);
1935 }
1936 }
1937 if (std::abs(coeff_mk) < coeff_cutoff) { ++skipped; continue; }
1938
1939 std::array<int, LDIM> k_copy = kidx;
1940 Function<T,LDIM> phi = FunctionFactory<T,LDIM>(world)
1941 .special_points({Vector<double,LDIM>(0.0)})
1942 .functor([k_copy, am](const Vector<double,LDIM>& r) -> T {
1943 double val = std::exp(-am * madness::inner(r, r));
1944 for (std::size_t d = 0; d < LDIM; ++d) {
1945 for (int p = 0; p < k_copy[d]; ++p) val *= r[d];
1946 }
1947 return T(val);
1948 });
1949 Y.push_back(phi);
1950 }
1951 }
1952 print("build_taylor_y: M =", M, "max_order =", max_order,
1953 "Y =", Y.size(), "(skipped", skipped, "below cutoff", coeff_cutoff, ")");
1954 return Y;
1955 }

1957 /// Generate all multi-indices k = (k_1, ..., k_LDIM) with total degree |k| <= max_order.
1958 static std::vector<std::array<int, LDIM>> generate_multi_indices(int max_order) {
1959 std::vector<std::array<int, LDIM>> result;
1960 std::array<int, LDIM> k{};
1961 std::function<void(std::size_t, int)> recurse = [&](std::size_t dim, int remaining) {
1962 if (dim == LDIM) {
1963 result.push_back(k);
1964 return;
1965 }
1966 for (int ki = 0; ki <= remaining; ++ki) {
1967 k[dim] = ki;
1968 recurse(dim + 1, remaining - ki);
1969 }
1970 };
1971 recurse(0, max_order);
1972 return result;
1973 }

1976 /// Construct LRF directly from the Gaussian expansion of an F12 operator.
1977 ///
1978 /// Instead of random-Y probing, uses the known separable structure:
1979 /// each Gaussian term c_m exp(-alpha_m |r1-r2|^2) is Taylor-expanded via
1980 /// exp(2 alpha_m r1.r2) = sum_k (2 alpha)^|k| r1^k r2^k / k!
1981 /// giving explicit g and h basis functions. After rr_cholesky rank
1982 /// reduction, yields a canonical LRF accurate to the operator's Gaussian
1983 /// fit precision (~1e-8), bypassing the O(1/r^{3/2}) error floor of
1984 /// random probing for Slater-type kernels.
1985 ///
1986 /// @param[in] functor the F12 functor with operator and a,b functions
1987 /// @param[in] target_error controls coefficient cutoff and rank reduction tol
1988 /// @param[in] max_taylor_order max total degree of Taylor expansion per Gaussian term
1989 /// @return canonical LowRankFunction
1990 LowRankFunction<T,NDIM> project_from_operator(
1991 const LRFunctorF12<T,NDIM>& functor,
1992 const double target_error = FunctionDefaults<NDIM>::get_thresh(),
1993 const int max_taylor_order = 15) const
1994 {
1995 World& world = functor.world();
1996 timer t1(world);
1997 t1.do_print = true;
1998
1999 double coeff_cutoff = target_error * target_error * 1.e-4;
2000 auto Y = build_taylor_y(functor, max_taylor_order, coeff_cutoff);
2001 t1.tag("direct: Y construction");
2002
2003 if (Y.empty()) {
2004 print("WARNING project_from_operator: no significant Y terms");
2005 return LowRankFunction<T,NDIM>();
2006 }
2007
2008 // 4. Use project_stable with the Taylor-constructed Y basis
2009 // This lets the SeparatedConvolution handle coefficient summation
2010 // correctly: h_raw = K*(a*Y) sums over ALL Gaussian terms, preserving
2011 // cancellations that the pure Taylor construction loses.
2012 double tol = std::max(1.e-14, std::min(1.e-3, target_error * target_error));
2013 ProjectionState empty;
2014 auto state = project_stable(functor, Y, tol, empty);
2015 auto lrf = LowRankFunction<T,NDIM>(state.g, state.h_ortho, tol);
2016
2017 double current_error = lrf.l2error_pythagoras(state.f_norm_sq);
2018 print("direct: Pythagoras error =", current_error, "rank =", state.rank());
2019 t1.tag("direct: projection");
2020
2021 // incremental tol-tightening if needed
2022 while (current_error > target_error && state.ovlp_h.size() > 0) {
2023 double tighter = std::max(1.e-14, tol * 0.01);
2024 if (tighter >= tol) break;
2025 auto state2 = project_stable(functor, Y, tighter, state);
2026 auto lrf2 = LowRankFunction<T,NDIM>(state2.g, state2.h_ortho, tighter);
2027 double error2 = lrf2.l2error_pythagoras(state2.f_norm_sq);
2028 print("direct: tol", tol, "->", tighter, ": error =", error2, "rank =", state2.rank());
2029 if (error2 >= current_error * 0.9) break; // stagnated
2030 lrf = lrf2;
2031 state = state2;
2032 current_error = error2;
2033 tol = tighter;
2034 t1.tag("direct: tol tightening");
2035 }
2036
2037 print("direct: final error =", current_error, "rank =", lrf.rank());
2038 return lrf;
2039 }

2042 /// apply a rhs on grid points to the hi-dim function and form Y = A_ij w_j (in Halko's language).
2043 /// The RHS selection (localized Gaussian/exponential vs harmonics) is controlled by
2044 /// the single `gridtype` parameter: values "random"/"adaptive"/"twostage" use
2045 /// localized Gaussian RHS (the previous "exponential" behavior), whereas
2046 /// "harmonics" uses the RI harmonics-based RHS.
2047 YFormationResult Yformer(const LRFunctorBase<T,NDIM>& lrfunctor1,
2048 const std::vector<Vector<double,LDIM>>& grid,
2049 const LowRankFunctionParameters& parameters,
2050 const double exponent=30.0,
2051 const double significance_tol=-1.0) const {

2055 if (parameters.gridtype()=="harmonics") { // use harmonics-based LHS

2059 std::vector<Function<double,LDIM>> omega;

2063 // .thresh(1.e-6) // make sure this is not undersampled
2064 .special_points({point})
2065 .functor([&point,&exponent,&coeff](const Vector<double,LDIM>& r)

2069 }));
2070 }
2071 Y=inner(lrfunctor1,omega,p2,p1);

2088 /// return a set of solid harmonic functions up to lmax with a zeta range on the given (atomic) centers
2089
2090 /// @param[in] World
2091 /// @param[in] zeta_range range of the Gaussian exponent zeta, given as {zeta_min, zeta_max, zeta_step}
2092 /// for even-tempered progression
2093 /// @param[in] lmax maximum angular momentum of the solid harmonics
2094 /// @param[in] centers the centers of the solid harmonics, typically the atomic positions
2095 static std::vector<Function<T, LDIM>> harmonic_basis(World& world,
2096 const std::vector<double> zeta_range, const int lmax,
2097 const std::vector<Vector<double, LDIM>> centers)
2098 {
2099 std::vector<Function<T, LDIM>> harmonics;
2100 // harmonics-based Y
2101 struct GaussianFunction : public FunctionFunctorInterface<double, LDIM>
2102 {
2103 std::vector<long> ijk; // angular momentum
2104 Vector<double, LDIM> center;
2105 double zeta = 1.0;
2106 GaussianFunction() = default;
2107
2108 GaussianFunction(const std::vector<long>& ijk, const Vector<double, LDIM>& center, const double zeta)
2109 : ijk(ijk), center(center), zeta(zeta)
2110 {
2111 }
2112
2113 double operator()(const Vector<double, LDIM>& r) const override
2114 {
2115 auto r_rel = r - center;
2116 double val = exp(-zeta * madness::inner(r_rel, r_rel));
2117 for (int d = 0; d < LDIM; ++d) val *= std::pow(r_rel[d], ijk[d]);
2118 return val;
2119 }
