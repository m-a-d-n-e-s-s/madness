1// operator_diagnostics.h — unified <ab|op|ij> diagnostic matrix container.
2//
3// Provides a map-based container (DiagnosticMatrix) and projection utilities:
4// project_xy — core primitive: result(a,b) += sum_k <bra[k].g[a](1) bra[k].h[b](2)|ket>
5// signature: project_xy(bra, ket)
6// project_ab — thin wrapper: bra = {CCPairFunction(ao_basis, ao_basis)}
7// project_Gab — thin wrapper: builds nfit Schwinger bra slots, calls project_xy
8//
9// Convention: `ref` = LDIM-D analytic reference, set by the caller.
10// `result` = NDIM-D projection, set by the methods below.
11// `error` = ||ref - result||_F, computed by compute_errors().
12//
13// Template parameters:
14// T — scalar type (default double)
15// NDIM — pair-space dimension (default 6); one-particle space is LDIM = NDIM/2.
16//
17// Bra representation: each element of the bra vector is a decomposed
18// CCPairFunction<T,NDIM> where get_a()[a] is the particle-1 factor for basis
19// index a and get_b()[b] is the particle-2 factor for basis index b. Weights
20// are absorbed into the get_a() functions by the bra builder.

37struct DiagnosticEntry {
38 Tensor<T> ref; ///< LDIM-D analytic reference (set by caller)
39 Tensor<T> result; ///< NDIM-D projection result (set by DiagnosticMatrix methods)
40 double error = -1.0;
41
42 void compute_error() {
43 if (ref.size() > 0 && result.size() > 0)
44 error = (ref - result).normf();
45 }

48/// Container for named <ab|op|ij> diagnostic matrix elements.
49///
50/// Each named entry holds a pair (ref, result) of nbasis×nbasis tensors and
51/// their Frobenius-norm difference. The caller fills `ref` from LDIM-D analytic
52/// formulas; the project_ab / project_Gab methods fill `result` by projecting
53/// NDIM-D functions onto the AO observer basis pairs |ab>.
54///
55/// @tparam T scalar type (default double)
56/// @tparam NDIM pair-space dimension (default 6); LDIM = NDIM/2 is the one-particle space
57template<typename T = double, std::size_t NDIM = 6>
58class DiagnosticMatrix {
59public:
60 static constexpr std::size_t LDIM = NDIM / 2; ///< one-particle space dimension
61
62 /// element provider: M(x,y) = <p1[x](1) p2[y](2) | X | ij> for arbitrary
63 /// LDIM-D bra lists p1, p2 — the operator-specific unit plugged into
64 /// ref_Gab / ref_GQab. Build via the provider factories of the operator
65 /// classes (CorrelationFactor::ue_*_provider, ExchangeCommutator::*_provider),
66 /// which hoist convolution operators and intermediates into the closure.
67 using ElementProvider = std::function<Tensor<T>(
68 const std::vector<Function<T,LDIM>>&,
69 const std::vector<Function<T,LDIM>>&)>;
70
71 std::map<std::string, DiagnosticEntry<T>> entries;
72 std::vector<Function<T,LDIM>> ao_basis; ///< orthonormalized observer basis
73 double time = 0.0;

75 /// Construct with a World reference and an AO basis.
76 /// If the basis is not orthonormal (||S - I||_F > 1e-10) it is
77 /// orthonormalized in-place via symmetric (Löwdin) orthonormalization.
78 /// Pass orthonormalize=false to keep the basis as given (e.g. when matrix
79 /// elements in the raw basis carry physical meaning, like pair energies).

131 /// ref(a,b) = Σ_n w_n <ã_n b̃_n | Q₁₂ X | ij> with Q₁₂ = (1-O₁)(1-O₂),
132 /// O = Σ_k |k_ket><k_bra| (plain sum, matching StrongOrthogonalityProjector).
133 /// Q₁₂ is expanded on the ket side as scalar contractions: per slot
134 /// A − S1·B − C·S2ᵀ + S1·D·S2ᵀ
135 /// with element blocks over p1 = {ã_a} ∪ {occ_bra_k}, p2 = {b̃_b} ∪ {occ_bra_l}
136 /// and overlaps S1(a,k) = <ã_a|occ_ket_k>, S2(b,l) = <b̃_b|occ_ket_l>.
137 /// No nearly-vanishing functions like Q(g_n⋆a) are ever formed — applying Q
138 /// to the bra functions suffers catastrophic cancellation when the convolved
139 /// observers lie nearly in the occupied space (see operator_diagnostics.md).

201/// Strong-orthogonality-projected pair-energy weights over an orthonormal
202/// observer basis {a}:
203/// W^Q(a,b) = 2<phi_i_bra (Q a)|g12|(Q b) phi_j_bra>
