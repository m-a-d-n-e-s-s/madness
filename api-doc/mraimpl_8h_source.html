91 template <typename T, std::size_t NDIM>
92 void FunctionCommonData<T,NDIM>::_init_quadrature
93 (int k, int npt, Tensor<double>& quad_x, Tensor<double>& quad_w,
94 Tensor<double>& quad_phi, Tensor<double>& quad_phiw, Tensor<double>& quad_phit) {
95 quad_x = Tensor<double>(npt); // point
96 quad_w = Tensor<double>(npt); // wheight
97 quad_phi = Tensor<double>(npt,k);
98 quad_phiw = Tensor<double>(npt,k);
99
100 gauss_legendre(npt,0.0,1.0,quad_x.ptr(),quad_w.ptr());
101 for (int mu=0; mu<npt; ++mu) {
102 double phi[200];
103 legendre_scaling_functions(quad_x(mu),k,phi);
104 for (int j=0; j<k; ++j) {
105 quad_phi(mu,j) = phi[j];
106 quad_phiw(mu,j) = quad_w(mu)*phi[j];
107 }
108 }
109 quad_phit = transpose(quad_phi);
110 }

120 template <typename T, std::size_t NDIM>
121 bool FunctionImpl<T,NDIM>::verify_parents_and_children() const {
122
123 // Ensure that parents and children exist appropriately
124 for (const auto& [key, node] : coeffs) {
125
126 if (key.level() > 0) {
127 const keyT parent = key.parent();
128 typename dcT::const_iterator pit = coeffs.find(parent).get();
129 if (pit == coeffs.end()) {
130 print(world.rank(), "FunctionImpl: verify: MISSING PARENT",key,parent);
131 std::cout.flush();
132 return false;
133 MADNESS_EXCEPTION("FunctionImpl: verify: MISSING PARENT", 0);
134 }
135 const nodeT& pnode = pit->second;
136 if (!pnode.has_children()) {
137 print(world.rank(), "FunctionImpl: verify: PARENT THINKS IT HAS NO CHILDREN",key,parent);
138 return false;
139 std::cout.flush();
140 MADNESS_EXCEPTION("FunctionImpl: verify: PARENT THINKS IT HAS NO CHILDREN", 0);
141 }
142 }
143
144 for (KeyChildIterator<NDIM> kit(key); kit; ++kit) {
145 typename dcT::const_iterator cit = coeffs.find(kit.key()).get();
146 if (cit == coeffs.end()) {
147 if (node.has_children()) {
148 print(world.rank(), "FunctionImpl: verify: MISSING CHILD",key,kit.key());
149 return false;
150 std::cout.flush();
151 MADNESS_EXCEPTION("FunctionImpl: verify: MISSING CHILD", 0);
152 }
153 }
154 else {
155 if (! node.has_children()) {
156 print(world.rank(), "FunctionImpl: verify: UNEXPECTED CHILD",key,kit.key());
157 return false;
158 std::cout.flush();
159 MADNESS_EXCEPTION("FunctionImpl: verify: UNEXPECTED CHILD", 0);
160 }
161 }
162 }
163 }
164 world.gop.fence();
165 return true;
166 }

170 template<typename T, std::size_t NDIM>
171 bool FunctionImpl<T,NDIM>::verify_tree_state_local() const {
172
173 const TreeState state=get_tree_state();
174 const int k=get_k();
175
176 auto check_internal_coeff_size = [&state, &k](const coeffT& c) {
177 if (state==compressed or state==nonstandard or state==nonstandard_with_leaves)
178 return c.dim(0)==2*k;
179 if (state==redundant) return c.dim(0)==k;
180 if (state==reconstructed) return (not c.is_assigned()); // must not be assigned at all
181 if (state==redundant_after_merge or state==nonstandard_after_apply) return true;
182 MADNESS_EXCEPTION("unknown state",1);
183 };
184 auto check_leaf_coeff_size = [&state, &k](const coeffT& c) {
185 // citation from compress_spawn
186 // if (not keepleaves) node.clear_coeff();
187 if (state==reconstructed or state==redundant or state==redundant_after_merge or state==nonstandard_with_leaves)
188 return c.dim(0)==k;
189 if (state==compressed or state==nonstandard) return (not c.is_assigned()); // must not be assigned at all
190 if (state==nonstandard_after_apply) return true;
191 MADNESS_EXCEPTION("unknown state",1);
192 };
193
194 bool good=true;
195 for (const auto& [key, node] : coeffs) {
196 const auto& c=node.coeff();
197 const bool is_internal=node.has_children();
198 const bool is_leaf=not node.has_children();
199 if (is_internal) good=good and check_internal_coeff_size(c);
200 if (is_leaf) good=good and check_leaf_coeff_size(c);
201 if (not good) {
202 print("incorrect size of coefficients for key",key,"state",state,c.dim(0));;
203 }
204 }
205 return good;
206 }

214 /// perform: this= alpha*f + beta*g, invoked by result
215
216 /// f and g are reconstructed, so we can save on the compress operation,
217 /// walk down the joint tree, and add leaf coefficients; effectively refines
218 /// to common finest level.
219 /// @param[in] alpha prefactor for f
220 /// @param[in] f first addend
221 /// @param[in] beta prefactor for g
222 /// @param[in] g second addend
223 /// @return nothing, but leaves this's tree reconstructed and as sum of f and g
224 template <typename T, std::size_t NDIM>
225 void FunctionImpl<T,NDIM>::gaxpy_oop_reconstructed(const double alpha, const implT& f,
226 const double beta, const implT& g, const bool fence) {
227
228 MADNESS_CHECK_THROW(f.is_reconstructed(), "gaxpy_oop_reconstructed: f is not reconstructed");
229 MADNESS_CHECK_THROW(g.is_reconstructed(), "gaxpy_oop_reconstructed: g is not reconstructed");
230
231 ProcessID owner = coeffs.owner(cdata.key0);
232 if (world.rank() == owner) {
233
234 CoeffTracker<T,NDIM> ff(&f);
235 CoeffTracker<T,NDIM> gg(&g);
236
237 typedef add_op coeff_opT;
238 coeff_opT coeff_op(ff,gg,alpha,beta);
239 typedef insert_op<T,NDIM> apply_opT;
240 apply_opT apply_op(this);
241
242 woT::task(world.rank(), &implT:: template forward_traverse<coeff_opT,apply_opT>,
243 coeff_op, apply_op, cdata.key0);
244
245 }
246 set_tree_state(reconstructed);
247 if (fence) world.gop.fence();
248 }

429 /// collect the data for a plot of the MRA structure locally on each node
430
431 /// @param[in] xaxis the x-axis in the plot (can be any axis of the MRA box)
432 /// @param[in] yaxis the y-axis in the plot (can be any axis of the MRA box)
433 /// @param[in] el2
434 template <typename T, std::size_t NDIM>
435 Tensor<double> FunctionImpl<T,NDIM>::print_plane_local(const int xaxis, const int yaxis, const coordT& el2) {
436 coordT x_sim;
437 user_to_sim<NDIM>(el2,x_sim);
438 x_sim[0]+=1.e-10;
439
440 // dimensions are: (# boxes)(hue, x lo left, y lo left, x hi right, y hi right)
441 Tensor<double> plotinfo(coeffs.size(),5);
442 long counter=0;
443
444 // loop over local boxes, if the fit, add the info to the output tensor
445 typename dcT::const_iterator end = coeffs.end();
446 for (typename dcT::const_iterator it=coeffs.begin(); it!=end; ++it) {
447 const keyT& key = it->first;
448 const nodeT& node = it->second;
449
450 // thisKeyContains ignores dim0 and dim1
451 if (key.thisKeyContains(x_sim,xaxis,yaxis) and node.is_leaf() and (node.has_coeff())) {
452
453 Level n=key.level();
454 Vector<Translation,NDIM> l=key.translation();
455 // get the diametral edges of the node in the plotting plane
456 double scale=std::pow(0.5,double(n));
457 double xloleft = scale*l[xaxis];
458 double yloleft = scale*l[yaxis];
459 double xhiright = scale*(l[xaxis]+1);
460 double yhiright = scale*(l[yaxis]+1);
461
462 // convert back to user coordinates
463 Vector<double,4> user;
464 user[0]=xloleft*FunctionDefaults<NDIM>::get_cell_width()[xaxis] + FunctionDefaults<NDIM>::get_cell()(xaxis,0);
465 user[2]=xhiright*FunctionDefaults<NDIM>::get_cell_width()[xaxis] + FunctionDefaults<NDIM>::get_cell()(xaxis,0);
466 user[1]=yloleft*FunctionDefaults<NDIM>::get_cell_width()[yaxis] + FunctionDefaults<NDIM>::get_cell()(yaxis,0);
467 user[3]=yhiright*FunctionDefaults<NDIM>::get_cell_width()[yaxis] + FunctionDefaults<NDIM>::get_cell()(yaxis,0);
468
469
470 // if ((xloleft<-5.0) or (yloleft<-5.0) or (xhiright>5.0) or (yhiright>5.0)) continue;
471 if ((user[0]<-5.0) or (user[1]<-5.0) or (user[2]>5.0) or (user[3]>5.0)) continue;
472
473 // do rank or do error
474 double color=0.0;
475 if (1) {
476
477 const double maxrank=40;
478 do_convert_to_color hue(maxrank,false);
479 color=hue(node.coeff().rank());
480 } else {
481
482 // Make quadrature rule of higher order
483 const int npt = cdata.npt + 1;
484 Tensor<double> qx, qw, quad_phi, quad_phiw, quad_phit;
485 FunctionCommonData<T,NDIM>::_init_quadrature(k+1, npt, qx, qw, quad_phi, quad_phiw, quad_phit);
486 do_err_box< FunctionFunctorInterface<T,NDIM> > op(this, this->get_functor().get(), npt, qx, quad_phit, quad_phiw);
487
488 do_convert_to_color hue(1000.0,true);
489 double error=op(it);
490 error=sqrt(error);//*pow(2,key.level()*6);
491 color=hue(error);
492 }
493
494 plotinfo(counter,0)=color;
495 plotinfo(counter,1)=user[0];
496 plotinfo(counter,2)=user[1];
497 plotinfo(counter,3)=user[2];
498 plotinfo(counter,4)=user[3];
499 ++counter;
500 }
501 }
502
503 // shrink the info
504 if (counter==0) plotinfo=Tensor<double>();
505 else plotinfo=plotinfo(Slice(0,counter-1),Slice(_));
506 return plotinfo;
507 }

568 /// return the keys of the local leaf boxes
569 template <typename T, std::size_t NDIM>
570 std::vector<typename FunctionImpl<T,NDIM>::keyT> FunctionImpl<T,NDIM>::local_leaf_keys() const {
571
572 // coeffs.size is maximum number of keys (includes internal keys)
573 std::vector<keyT> keys(coeffs.size());
574
575 // loop over local boxes, if they are leaf boxes add their quadrature roots
576 // to the output tensor
577 int i=0;
578 typename dcT::const_iterator end = coeffs.end();
579 for (typename dcT::const_iterator it=coeffs.begin(); it!=end; ++it) {
580 const keyT& key = it->first;
581 const nodeT& node = it->second;
582 if (node.is_leaf()) keys[i++]=key;
583 }
584
585 // shrink the vector to number of leaf keys
586 keys.resize(i);
587 return keys;
588 }

590 /// print the grid in xyz format
591
592 /// the quadrature points and the key information will be written to file,
593 /// @param[in] filename where the quadrature points will be written to

659 /// Returns the truncation threshold according to truncate_method
660 template <typename T, std::size_t NDIM>
661 double FunctionImpl<T,NDIM>::truncate_tol(double tol, const keyT& key) const {
662
663 // RJH ... introduced max level here to avoid runaway
664 // refinement due to truncation threshold going down to
665 // intrinsic numerical error
666 const int MAXLEVEL1 = 20; // 0.5**20 ~= 1e-6
667 const int MAXLEVEL2 = 10; // 0.25**10 ~= 1e-6
668
669 if (truncate_mode == 0) {
670 return tol;
671 }
672 else if (truncate_mode == 1) {
673 double L = FunctionDefaults<NDIM>::get_cell_min_width();
674 return tol*std::min(1.0,pow(0.5,double(std::min(key.level(),MAXLEVEL1)))*L);
675 }
676 else if (truncate_mode == 2) {
677 double L = FunctionDefaults<NDIM>::get_cell_min_width();
678 return tol*std::min(1.0,pow(0.25,double(std::min(key.level(),MAXLEVEL2)))*L*L);
679 }
680 else if (truncate_mode == 3) {
681 // similar to truncate mode 1, but with an additional factor to
682 // account for an increased number of boxes in higher dimensions
683
684 // here is our handwaving argument: this threshold will give each
685 // FunctionNode an error of less than tol. The total error can
686 // then be as high as sqrt(#nodes) * tol. Therefore in order to
687 // account for higher dimensions: divide tol by about the root of
688 // number of siblings (2^NDIM) that have a large error when we
689 // refine along a deep branch of the tree. FAB
690 //
691 // Nope ... it can easily be as high as #nodes * tol. The real
692 // fix for this is an end-to-end error analysis of the larger
693 // application and if desired to include this factor into the
694 // threshold selected by the application. RJH
695 const static double fac=1.0/std::pow(2,NDIM*0.5);
696 tol*=fac;
697
698 double L = FunctionDefaults<NDIM>::get_cell_min_width();
699 return tol*std::min(1.0,pow(0.5,double(std::min(key.level(),MAXLEVEL1)))*L);
700
701 } else {
702 MADNESS_EXCEPTION("truncate_mode invalid",truncate_mode);
703 }
704 }

716 /// Directly project parent NS coeffs to child NS coeffs
717
718 template <typename T, std::size_t NDIM>
719 typename FunctionImpl<T,NDIM>::coeffT FunctionImpl<T,NDIM>::parent_to_child_NS(
720 const keyT& child, const keyT& parent, const coeffT& coeff) const {
721
722 const implT* f=this;
723 // MADNESS_ASSERT(coeff.tensor_type()==TT_FULL);
724 coeffT result(f->cdata.v2k,coeff.tensor_type());
725
726 // if the node for child is existent in f, and it is an internal node, we
727 // automatically have the NS form; if it is a leaf node, we only have the
728 // sum coeffs, so we take zero difference coeffs
729 if (child==parent) {
730 if (coeff.dim(0)==2*f->get_k()) result=coeff; // internal node
731 else if (coeff.dim(0)==f->get_k()) { // leaf node
732 result(f->cdata.s0)+=coeff;
733 } else {
734 MADNESS_EXCEPTION("confused k in parent_to_child_NS",1);
735 }
736 } else if (child.level()>parent.level()) {
737
738 // parent and coeff should refer to a leaf node with sum coeffs only
739 // b/c tree should be compressed with leaves kept.
740 MADNESS_ASSERT(coeff.dim(0)==f->get_k());
741 const coeffT scoeff=f->parent_to_child(coeff,parent,child);
742 result(f->cdata.s0)+=scoeff;
743 } else {
744 MADNESS_EXCEPTION("confused keys in parent_to_child_NS",1);
745 }
746 return result;
747 }

775 /// Refine multiple functions down to the same finest level
776
777 /// @param[v] is the vector of functions we are refining.
778 /// @param[key] is the current node.
779 /// @param[c] is the vector of coefficients passed from above.
780 template <typename T, std::size_t NDIM>
781 void FunctionImpl<T,NDIM>::refine_to_common_level(const std::vector<FunctionImpl<T,NDIM>*>& v,
782 const std::vector<tensorT>& c,
783 const keyT key) {
784 if (key == cdata.key0 && coeffs.owner(key)!=world.rank()) return;
785
786 // First insert coefficients from above ... also get write accessors here
787 std::unique_ptr<typename dcT::accessor[]> acc(new typename dcT::accessor[v.size()]);
788 for (unsigned int i=0; i<c.size(); i++) {
789 MADNESS_ASSERT(v[i]->coeffs.get_pmap() == coeffs.get_pmap());
790 MADNESS_ASSERT(v[i]->coeffs.owner(key) == world.rank());
791 bool exists = ! v[i]->coeffs.insert(acc[i],key);
792 if (c[i].size()) {
793 MADNESS_CHECK(!exists);
794 acc[i]->second = nodeT(coeffT(c[i],targs),false);
795 }
796 else {
797 MADNESS_ASSERT(exists);
798 }
799 }
800
801 // If everyone has coefficients we are done
802 bool done = true;
803 for (unsigned int i=0; i<v.size(); i++) {
804 done &= acc[i]->second.has_coeff();
805 }
806
807 if (!done) {
808 // Those functions with coefficients need to be refined down
809 std::vector<tensorT> d(v.size());
810 for (unsigned int i=0; i<v.size(); i++) {
811 if (acc[i]->second.has_coeff()) {
812 tensorT s(cdata.v2k);
813 // s(cdata.s0) = acc[i]->second.coeff()(___);
814 s(cdata.s0) = acc[i]->second.coeff().full_tensor();
815 acc[i]->second.clear_coeff();
816 d[i] = unfilter(s);
817 acc[i]->second.set_has_children(true);
818 }
819 }
820
821 // Loop thru children and pass down
822 for (KeyChildIterator<NDIM> kit(key); kit; ++kit) {
823 const keyT& child = kit.key();
824 std::vector<Slice> cp = child_patch(child);
825 std::vector<tensorT> childc(v.size());
826 for (unsigned int i=0; i<v.size(); i++) {
827 if (d[i].size()) childc[i] = copy(d[i](cp));
828 }
829 woT::task(coeffs.owner(child), &implT::refine_to_common_level, v, childc, child);
830 }
831 }
832 }

1028
1029 /// the result looks like: (v(1,2) + v(1) + v(2)) |ket(1,2)>
1030 /// or (v(1,2) + v(1) + v(2)) |p(1) p(2)>
1031 /// i.e. coefficients for the ket and coefficients for the two particles are
1032 /// mutually exclusive. All potential terms are optional, just pass in empty coeffs.
1033 /// @param[in] key the key of the FunctionNode to which these coeffs belong
1034 /// @param[in] cket coefficients of the ket
1035 /// @param[in] vpotential1 function values of the potential for particle 1
1036 /// @param[in] vpotential2 function values of the potential for particle 2
1037 /// @param[in] veri function values for the 2-particle potential
1038 template <typename T, std::size_t NDIM>
1039 typename FunctionImpl<T,NDIM>::coeffT FunctionImpl<T,NDIM>::assemble_coefficients(
1040 const keyT& key, const coeffT& coeff_ket, const coeffT& vpotential1,
1041 const coeffT& vpotential2, const tensorT& veri) const {
1042
1043 // take a shortcut if we are already done
1044 bool ket_only=(not (vpotential1.has_data() or vpotential2.has_data() or veri.has_data()));
1045 if (ket_only) return coeff_ket;
1046
1047 // switch to values instead of coefficients
1048 coeffT val_ket=coeffs2values(key,coeff_ket);
1049
1050 // the result tensor
1051 coeffT val_result=coeffT(val_ket.ndim(),val_ket.dims(),this->get_tensor_args().tt);
1052 coeffT coeff_result;
1053
1054 // potential for particles 1 and 2, must be done in TT_2D
1055 if (vpotential1.has_data() or vpotential2.has_data()) {
1056 val_ket=val_ket.convert(TensorArgs(-1.0,TT_2D));
1057 }
1058 if (vpotential1.has_data()) val_result+=multiply(val_ket,vpotential1,0);
1059 if (vpotential2.has_data()) val_result+=multiply(val_ket,vpotential2,1);
1060
1061 // values for eri: this must be done in full rank...
1062 if (veri.has_data()) {
1063 tensorT val_ket2=val_ket.full_tensor_copy().emul(veri);
1064 if (val_result.has_data()) val_ket2+=val_result.full_tensor();
1065 // values2coeffs expensive (30%), coeffT() (relatively) cheap (8%)
1066 coeff_result=coeffT(values2coeffs(key,val_ket2),this->get_tensor_args());
1067
1068 } else {
1069
1070 // convert back to original tensor type
1071 val_ket=val_ket.convert(get_tensor_args());
1072 MADNESS_ASSERT(val_result.has_data());
1073 coeff_result=values2coeffs(key,val_result);
1074 coeff_result.reduce_rank(this->get_tensor_args().thresh);
1075 }
1076
1077 return coeff_result;
1078
1079 }

1331 void FunctionImpl<T,NDIM>::broaden(const array_of_bools<NDIM>& is_periodic, bool fence) {
1332 typename dcT::iterator end = coeffs.end();

1334 const keyT& key = it->first;
1335 typename dcT::accessor acc;

1337 MADNESS_CHECK(found);
1338 nodeT& node = acc->second;
1339 if (node.has_coeff() &&
1340 node.get_norm_tree() != -1.0 &&
1341 node.coeff().normf() >= truncate_tol(thresh,key)) {

1353 const int odd = key.translation()[d] & 0x1L; // 1 if odd, 0 if even

1370 }

1393 void FunctionImpl<T,NDIM>::trickle_down_op(const keyT& key, const coeffT& s) {
1394 // Note that after application of an integral operator not all
1395 // siblings may be present so it is necessary to check existence
1396 // and if absent insert an empty leaf node.
1397 //
1398 // If summing the result of an integral operator (i.e., from
1399 // non-standard form) there will be significant scaling function
1400 // coefficients at all levels and possibly difference coefficients
1401 // in leaves, hence the tree may refine as a result.
1402 typename dcT::iterator it = coeffs.find(key).get();
1403 if (it == coeffs.end()) {
1404 coeffs.replace(key,nodeT(coeffT(),false));
1405 it = coeffs.find(key).get();
1406 }
1407 nodeT& node = it->second;
1408
1409 // The integral operator will correctly connect interior nodes
1410 // to children but may leave interior nodes without coefficients
1411 // ... but they still need to sum down so just give them zeros
1412 if (node.coeff().has_no_data()) node.coeff()=coeffT(cdata.vk,targs);
1413
1414 // if (node.has_children() || node.has_coeff()) { // Must allow for inconsistent state from transform, etc.
1415 if (node.has_children()) { // Must allow for inconsistent state from transform, etc.
1416 coeffT d = node.coeff();
1417 if (key.level() > 0) d += s; // -- note accumulate for NS summation
1418 node.clear_coeff();
1419 for (KeyChildIterator<NDIM> kit(key); kit; ++kit) {
1420 const keyT& child = kit.key();
1421 coeffT ss= upsample(child,d);
1422 ss.reduce_rank(thresh);
1423 PROFILE_BLOCK(recon_send);

1433 /// change the tree state of this function, might or might not respect fence!
1434 template <typename T, std::size_t NDIM>
1435 void FunctionImpl<T,NDIM>::change_tree_state(const TreeState finalstate, bool fence) {
1436
1437 TreeState current_state=get_tree_state();
1438 if (current_state==finalstate) return;
1439
1440 // try direct conversion if possible, otherwise change to reconstructed state,
1441 // and then convert to final state
1442
1443 // try direct conversion
1444 if (finalstate==reconstructed) { // this MUST cover all cases
1445 if (current_state==compressed) reconstruct(fence);
1446 else if (current_state==nonstandard) reconstruct(fence);
1447 else if (current_state==nonstandard_after_apply) reconstruct(fence);
1448 else if (current_state==nonstandard_with_leaves) {
1449 remove_internal_coefficients(fence);
1450 set_tree_state(reconstructed);
1451 }
1452 else if (current_state==redundant) {
1453 remove_internal_coefficients(fence);
1454 set_tree_state(reconstructed);
1455 }
1456 else if (current_state==redundant_after_merge) {
1457 sum_down(fence);
1458 set_tree_state(reconstructed);
1459 }
1460 else if (current_state==redundant_after_merge) sum_down(fence);
1461 else MADNESS_EXCEPTION("unknown/unsupported current tree state",1);
1462 set_tree_state(reconstructed);
1463 } else if (finalstate==compressed) { // cases that are not covered will be done in two steps
1464 if (current_state==reconstructed) compress(compressed,fence);
1465 if (current_state==nonstandard) standard(fence);
1466 if (current_state==nonstandard_with_leaves) standard(fence);
1467 } else if (finalstate==nonstandard) { // cases that are not covered will be done in two steps
1468 if (current_state==reconstructed) compress(nonstandard,fence);
1469 if (current_state==nonstandard_with_leaves) {
1470 remove_leaf_coefficients(fence);
1471 set_tree_state(nonstandard);
1472 }
1473 } else if (finalstate==nonstandard_with_leaves) { // cases that are not covered will be done in two steps
1474 if (current_state==reconstructed) compress(nonstandard_with_leaves,fence);
1475 } else if (finalstate==redundant) { // cases that are not covered will be done in two steps
1476 if (current_state==reconstructed) make_redundant(fence);
1477 } else {
1478 MADNESS_EXCEPTION("unknown/unsupported final tree state",1);
1479 }
1480 if (fence && VERIFY_TREE) verify_tree(); // Must be after in case nonstandard
1481
1482 // direct conversion worked, we're good
1483 if (finalstate==get_tree_state()) return;
1484
1485
1486 // go through reconstructed state -- requires fence! The uncovered transitions need the
1487 // reconstructed coefficients before the second pass, so the fence is unavoidable; report a
1488 // broken no-fence promise once, from one rank, rather than per rank per call.
1489 if (not fence and FunctionDefaults<NDIM>::get_debug() and world.rank() == 0) {
1490 static std::atomic<bool> reported(false);
1491 if (not reported.exchange(true))
1492 print("change_tree_state:", current_state, "->", finalstate,
1493 "must reconstruct first and therefore fences, despite fence=false");
1494 }
1495 change_tree_state(reconstructed,true);
1496 change_tree_state(finalstate,fence);
1497
1498 }

1648 /// given the sum coefficients of all children, truncate or not
1649
1650 /// @return new sum coefficients (empty if internal, not empty, if new leaf); might delete its children
1651 template <typename T, std::size_t NDIM>
1652 typename FunctionImpl<T,NDIM>::coeffT FunctionImpl<T,NDIM>::truncate_reconstructed_op(const keyT& key, const std::vector< Future<coeffT > >& v, const double tol) {
1653
1654 MADNESS_ASSERT(coeffs.probe(key));
1655
1656 // the sum coefficients might be empty, which means they come from an internal node
1657 // and we must not truncate; so just return empty coeffs again
1658 for (size_t i=0; i<v.size(); ++i) if (v[i].get().has_no_data()) return coeffT();
1659
1660 // do not truncate below level 1
1661 if (key.level()<2) return coeffT();
1662
1663 // compute the wavelet coefficients from the child nodes

2089 template <typename T, std::size_t NDIM>
2090 T FunctionImpl<T,NDIM>::eval_cube(Level n, coordT& x, const tensorT& c) const {
2091 PROFILE_MEMBER_FUNC(FunctionImpl);
2092 const int k = cdata.k;
2093 T sum = T(0.0);
2094
2095 // v = sum_{p,q,...} c[p,q,...] phi_p(x0) phi_q(x1) ... is a separable
2096 // contraction; the fastest evaluation depends on the dimension.
2097 //
2098 // NDIM<=2 (deep 1-D radial trees are the hottest eval workload): a
2099 // factored register-resident loop. Partial sums stay in registers, the
2100 // only memory traffic is one streaming read of c, and no per-thread
2101 // scratch is needed (px is <= NDIM*MAXK*8 bytes of stack). Routing the
2102 // tiny (k x 1) contraction through general_fast_transform's dispatch
2103 // and ping-pong scratch measured ~20% slower at NDIM=1.
2104 //
2105 // NDIM>=3: general_fast_transform's staged contraction vectorizes
2106 // (the factored loop's inner reduction cannot under strict FP) and
2107 // measured 4-5x faster at NDIM=6; the phi matrices and scratch are
2108 // thread_local, so this path stays allocation-free after warm-up.
2109 if constexpr (NDIM <= 2) {
2110 MADNESS_ASSERT(k <= MAXK);
2111 double px[NDIM][MAXK];
2112 for (std::size_t i=0; i<NDIM; ++i) legendre_scaling_functions(x[i],k,px[i]);
2113
2114 if constexpr (NDIM == 1) {
2115 const T* cp = c.ptr();
2116 for (int p=0; p<k; ++p) sum += cp[p]*px[0][p];
2117 }
2118 else {
2119 for (int p=0; p<k; ++p) {
2120 const double a = px[0][p];
2121 const T* cq = &c(p,0);
2122 T s2 = T(0);
2123 for (int q=0; q<k; ++q) s2 += cq[q]*px[1][q];
2124 sum += a*s2;
2125 }
2126 }
2127 }
2128 else {
2129 thread_local Tensor<double> phi[NDIM];
2130 thread_local int phi_k = -1;
2131 if (phi_k != k) {
2132 for (std::size_t i=0; i<NDIM; ++i) phi[i] = Tensor<double>(long(k), 1L);
2133 phi_k = k;
2134 }
2135 for (std::size_t i=0; i<NDIM; ++i)
2136 legendre_scaling_functions(x[i], k, phi[i].ptr());
2137
2138 typedef TENSOR_RESULT_TYPE(T,double) evalR;
2139 // ws/res are references bound to the thread_local scratch tensors.
2140 auto [ws, res] = madness::detail::eval_scratch<evalR>(c.size());
2141 general_fast_transform(c, phi, res, ws);
2142 sum = res.ptr()[0];
2143 }
2144 // exp2 replaces pow for the level scaling; 1/sqrt(cell_volume) is cached
2145 // per thread and refreshed only if the cell changes (perf-doc change #2).
2146 thread_local double cached_cell_volume = -1.0;
2147 thread_local double cached_inv_sqrt_cell_vol = 0.0;
2148 const double cell_volume = FunctionDefaults<NDIM>::get_cell_volume();
2149 if (cell_volume != cached_cell_volume) {
2150 cached_cell_volume = cell_volume;
2151 cached_inv_sqrt_cell_vol = 1.0/std::sqrt(cell_volume);
2152 }
2153 return sum * std::exp2(0.5*NDIM*n) * cached_inv_sqrt_cell_vol;
2154 }

2156 template <typename T, std::size_t NDIM>
2157 void FunctionImpl<T,NDIM>::reconstruct_op(const keyT& key, const coeffT& s, const bool accumulate_NS) {
2158 //PROFILE_MEMBER_FUNC(FunctionImpl);
2159 // Note that after application of an integral operator not all
2160 // siblings may be present so it is necessary to check existence
2161 // and if absent insert an empty leaf node.
2162 //
2163 // If summing the result of an integral operator (i.e., from
2164 // non-standard form) there will be significant scaling function
2165 // coefficients at all levels and possibly difference coefficients
2166 // in leaves, hence the tree may refine as a result.
2167 typename dcT::iterator it = coeffs.find(key).get();
2168 if (it == coeffs.end()) {
2169 coeffs.replace(key,nodeT(coeffT(),false));
2170 it = coeffs.find(key).get();
2171 }
2172 nodeT& node = it->second;
2173
2174 // The integral operator will correctly connect interior nodes
2175 // to children but may leave interior nodes without coefficients
2176 // ... but they still need to sum down so just give them zeros
2177 if (node.has_children() && !node.has_coeff()) {
2178 node.set_coeff(coeffT(cdata.v2k,targs));
2179 }
2180
2181 if (node.has_children() || node.has_coeff()) { // Must allow for inconsistent state from transform, etc.
2182 coeffT d = node.coeff();
2183 if (!d.has_data()) d = coeffT(cdata.v2k,targs);
2184 if (accumulate_NS and (key.level() > 0)) d(cdata.s0) += s; // -- note accumulate for NS summation
2185 if (d.dim(0)==2*get_k()) { // d might be pre-truncated if it's a leaf
2186 d = unfilter(d);
2187 node.clear_coeff();
2188 node.set_has_children(true);
2189 for (KeyChildIterator<NDIM> kit(key); kit; ++kit) {
2190 const keyT& child = kit.key();
2191 coeffT ss = copy(d(child_patch(child)));
2192 ss.reduce_rank(thresh);
2193 //PROFILE_BLOCK(recon_send); // Too fine grain for routine profiling
2194 woT::task(coeffs.owner(child), &implT::reconstruct_op, child, ss, accumulate_NS);
2195 }
2196 } else {
2197 MADNESS_ASSERT(node.is_leaf());
2198 // node.coeff()+=s;
2199 node.coeff().reduce_rank(targs.thresh);
2200 }
2201 }
2202 else {
2203 coeffT ss=s;
2204 if (s.has_no_data()) ss=coeffT(cdata.vk,targs);
2205 if (key.level()) node.set_coeff(copy(ss));
2206 else node.set_coeff(ss);
2207 }
