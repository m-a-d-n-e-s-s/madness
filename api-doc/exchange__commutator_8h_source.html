27 /// Uniform result envelope for every algorithm path. Each "piece" may
28 /// be empty when the algorithm only produces the combined commutator
29 /// (e.g. lrf-split-alpha constructs KffK directly).
30 struct KffKResult {
31 std::vector<CCPairFunction<double, 6>> Kf; ///< K̂ f₁₂ | i j ⟩

35 double mem_gb = 0.0; ///< combined size of pair factors (GByte)
36 long rank = 0; ///< combined LRF rank (0 for 6D)

49 /// Options for the three-range LRF k-commutator variant.
50 /// Partition the Coulomb GFit into:
51 /// * diffuse (α_μ < alpha_lo): wide Gaussians, large-ve LRF, low rank
52 /// * medium (alpha_lo ≤ α_μ ≤ alpha_hi): tighter LRF
53 /// * tight (α_μ > alpha_hi): discard (δ-cancellation in commutator)
54 /// See lrf_three_range_gfit.md for the derivation.
55 struct ThreeRangeOptions {
56 double alpha_lo = 1.0; ///< diffuse / medium boundary
57 double alpha_hi = 1.0e4; ///< medium / tight boundary (≡ alpha_star)
58 double lo = 1.0e-6;
59 double hi = 10.0;
60 double eps_gfit = 1.0e-6;
61 /// Per-range LRF construction parameters; both default-constructed.
62 /// Caller is expected to set radius / volume_element / tol / lmax /
63 /// f12type per range before invocation.
64 LowRankFunctionParameters lrfparam_diffuse;
65 LowRankFunctionParameters lrfparam_medium;
66 /// If true, build the medium-range LRF via project_from_operator
67 /// (Taylor / direct construction) instead of random-Y project().
68 /// Mandatory for the medium slab in isolation: random-Y stalls at
69 /// large error without diffuse anchor functions.
70 bool medium_use_taylor = true;
71 /// Max Taylor order for project_from_operator (passed straight
72 /// through; default matches lowrankfunction.h's own default).
73 int max_taylor_order = 15;
74 /// If true, the medium slab is treated by a *6D* apply with a
75 /// partial-Coulomb operator (only the medium Gaussians) instead
76 /// of any LRF. The diffuse slab still uses the LRF path.
77 /// Mutually exclusive with `medium_use_taylor`; takes precedence.
78 bool medium_use_6d = false;
79 };

111 /// Assemble the 6D exchange commutator [K̂, f] |ij⟩ using the split-α
112 /// LRF k-commutator algorithm. Sums over occupied orbitals in
113 /// info.mo_bra/info.mo_ket. Currently assumes phi_i == phi_j
114 /// (symmetric case) and exploits swap_particles for the K̂₂ piece.
115 ///
116 /// Output pair layout:
117 /// result.pair[0] = Σ_k Σ_ρ g_kρ(1) · (phi_j · A_kρ)(2) (separable)
118 /// result.pair[1] = Σ_k Σ_ρ B_kρ · (g_kρ·phi_i)(1) · f(1,2) · phi_j(2)
119 /// with coefficient −1
120 /// plus their particle-2 mirrors when i == j.

124 const CCFunction<double, 3>& phi_j,
125 const Info& info,
126 const LowRankFunction<double,6>& exchange_op,

170 // ---------------------------------------------------------------------
171 // G·[K̂,f] diagnostics — Schwinger-quadrature 3D reference vs 6D projection
172 // ---------------------------------------------------------------------
173
174 /// Diagnose ⟨ab | G [K̂,f₁₂] | ij⟩ by comparing a 6D projection against
175 /// a 3D Schwinger eigentime quadrature.
176 /// Returns a DiagnosticMatrix with entries "GKf", "GfK", "GKffK".
177 /// ref=3D Schwinger quadrature, result=6D projection <ab|G·piece|ij>, error=||ref-result||.
178 ///
179 /// The 6D references GKf_cc and GfK_cc are the G-applied commutator pieces
180 /// as CCPairFunction vectors. Accepting CCPairFunction (instead of a bare
181 /// real_function_6d) allows any algorithm variant to be tested:
182 ///
183 /// * 6D path (apply_KffK_6d): Kf and fK are pure-6D; caller applies G6d
184 /// via apply(G6d, kffk.Kf[0].get_function()) and wraps in a vector.
185 /// * LRF path (apply_KffK_lowrank_split_alpha, three_range): Kf may be
186 /// decomposed. Caller converts to pure (sum outer products), applies G6d,
187 /// and passes the result wrapped in a CCPairFunction vector.
188 /// * Mixed/multi-piece results: sum all pieces (each entry is summed inside).
189 ///
190 /// The 6D reference is projected using the same partial_inner / matrix_inner
191 /// machinery as diagnose(), handling both pure and decomposed CCPairFunctions.
192 ///
193 /// The 3D Schwinger formulas use only K̂, K̂†, and the Slater f₁₂:
194 ///
195 /// G Kf: ⟨ã b̃ | K̂₁ f₁₂ | φᵢ φⱼ⟩ = inner((K̂†ã)·φᵢ, f₁₂(b̃·φⱼ)) + K̂₂ swap
196 /// G fK: ⟨ã b̃ | f₁₂ K̂₁ | φᵢ φⱼ⟩ = inner(ã·K̂φᵢ, f₁₂(b̃·φⱼ)) + K̂₂ swap
197 ///
198 /// where ã = g_{αₙ}*a and b̃ = g_{αₙ}*b are Gaussian-smoothed AO basis
199 /// functions and the sum over quadrature nodes n (with 6D weight
200 /// w_n = c_n^{bsh}·(αₙ/π)^{3/2}) approximates G.
201 ///
202 /// NOTE: diagnose_GKffK and diagnose_GQKffK share the same signature.
203 /// @param GKf_cc G·K̂f₁₂|ij⟩ pieces (caller applies G6d externally)
204 /// @param GfK_cc G·f₁₂K̂|ij⟩ pieces
205 /// @param Kphi_i K̂φᵢ (precomputed once by caller)
206 /// @param Kphi_j K̂φⱼ (precomputed once by caller)
207 /// @param info molecular/orbital info carrying mo_ket, mo_bra, parameters
208 /// @param energy ε_i + ε_j (must be negative)
209 /// @param aobasis observer basis
210 /// @param orthonormalize_basis Löwdin-orthonormalize aobasis (false: keep raw)
211 DiagnosticMatrix<> diagnose_GKffK(
212 World& world,
213 const std::vector<CCPairFunction<double,6>>& GKf_cc,
214 const std::vector<CCPairFunction<double,6>>& GfK_cc,
215 const real_function_3d& phi_i,

228 /// Diagnose ⟨ab | G Q₁₂ [K̂,f₁₂] | ij⟩, Q₁₂ = (1-O₁)(1-O₂), O = Σ|k_ket⟩⟨k_bra|.
229 ///
230 /// Caller supplies G·Q₁₂·K̂f₁₂|ij⟩ and G·Q₁₂·f₁₂K̂|ij⟩ (neither G nor Q is
