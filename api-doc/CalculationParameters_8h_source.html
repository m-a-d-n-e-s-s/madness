300 void set_derived_values(const Molecule& molecule) {
301
302 for (size_t iatom = 0; iatom < molecule.natom(); iatom++) {
303 if (molecule.get_pseudo_atom(iatom)){
304 set_derived_value("pure_ae",false);
305 continue;
306 }
307 }
308 set_derived_value("aobasis",molecule.guess_file());;
309 const int n_core = molecule.n_core_orb_all();
310
311
312 std::vector<double> proto=get<std::vector<double> >("protocol");
313 // No ... The accuracy of computation is INDEPENDENT of the convergence requirement
314 // --- actually need more precision than convergence threshold in order to have
315 // variational principle working and for robust convergence
316 //proto.back()=get<double>("econv");
317 set_derived_value("protocol",proto);
318 // No ... the energy is variational! Don't need more accuracy in dconv --- in fact the opposite is true.
319 // set_derived_value("dconv",sqrt(get<double>("econv"))*0.1);
320
321 double z = molecule.total_nuclear_charge();
322 const double charge=get<double>("charge");
323 int nelec = int(z - charge - n_core*2);
324 if (fabs(nelec+charge+n_core*2-z) > 1e-6) {
325 error("non-integer number of electrons?", nelec+charge+n_core*2-z);
326 }
327
328 set_derived_value("nalpha",(nelec + nopen())/2);
329 set_derived_value("nbeta",(nelec - nopen())/2);
330
331 if (nalpha() < 0) error("negative number of alpha electrons?", nalpha());
332 if (nbeta() < 0) error("negative number of beta electrons?", nbeta());
333 if ((nalpha()+nbeta()) != nelec) error("nalpha+nbeta != nelec", nalpha()+nbeta());
334 if (nalpha() != nbeta()) set_derived_value("spin_restricted",false);
335
336 set_derived_value("nmo_alpha",nalpha() + nvalpha());
337 set_derived_value("nmo_beta",nbeta() + nvbeta());
338
339 if (nv_extra() < 0 or nv_step() < 0 or nv_its() < 1) error("nv_extra, nv_step >= 0 and nv_its >= 1 required");
340 if (nv_extra() > 0 and nvalpha() == 0) error("nv_extra requires nvalpha > 0");
341 if (freeze_occupied() and nvalpha() == 0 and nvbeta() == 0) error("freeze_occupied requires nvalpha or nvbeta > 0");
342
343 // Explicit occupations address the occupied span. A hole there is only
344 // index-stable with canonical orbitals: every iteration re-sorts them by
345 // eigenvalue, so the empty index stays on the same orbital.
346 bool hole = false;
347 for (const auto& [name, n] : std::vector<std::pair<std::string, int>>{{"aocc", nalpha()}, {"bocc", nbeta()}}) {
348 const auto occ = get<std::vector<double> >(name);
349 if (int(occ.size()) > n) error((name + " has more entries than occupied orbitals").c_str(), occ.size());
350 for (const double o : occ) {
351 if (o != 0.0 and o != 1.0) error((name + " entries must be 0 or 1").c_str(), o);
352 if (o == 0.0) hole = true;
353 }
354 }
355 if (not bocc().empty() and spin_restricted()) error("bocc requires spin_restricted false");
356 if (hole) {
357 set_derived_value("localize", std::string("canon"));
358 if (do_localize()) error("a hole in the explicit occupations requires localize canon");
359 }
360
361 // Unless overridden by the user use a cell big enough to
362 // have exp(-sqrt(2*I)*r) decay to 1e-6 with I=1ev=0.037Eh
363 // --> need 50 a.u. either side of the molecule
364 set_derived_value("l",molecule.bounding_cube() + 50.0);
365
366 //set_derived_value("lo",molecule.smallest_length_scale());
367
368 // set highest possible point group for symmetry
369 if (do_localize()) set_derived_value("pointgroup",std::string("c1"));
370 else set_derived_value("pointgroup",molecule.get_pointgroup());
371
372 // above two lines will not override user input, so check input is sane
373 if (do_localize() and do_symmetry()) {
374 error("\n\nsymmetry and localization cannot be used at the same time\n"
375 "switch from local to canonical orbitals (keyword canon)\n\n");
376 }
377
378 // Geometry optimization left the `dft` group entirely. `gopt` in particular
379 // must not be waved through: a deck that sets it expects an optimized
380 // geometry back, and silently returning a single point at the input
381 // geometry is a wrong answer, not a missing feature. They stay registered
382 // and erroring rather than being deleted, because ignore_unknown_keys is
383 // true by default and a deleted key would run the deck as a plain single
384 // point after one warning nobody reads.
385 for (const char* key : {"gopt","gtol","gtest","gval","gprec",
386 "gmaxiter","ginitial_hessian","algopt"}) {
387 if (is_user_defined(key))
388 error(("\n\n`" + std::string(key) + "` has been retired: geometry optimization is now a task of "
389 "its own.\nUse `madqc --optimize --wf=<scf|nemo>` and the `optimization` "
390 "parameter group\n(see `madqc --print_parameters=optimization`).\n\n").c_str());
391 }
392
393 // Retired keyvals. A clean break needs to be loud: ignore_unknown_keys is
394 // true by default, so deleting these outright would let an old deck run
395 // with different semantics after one warning nobody reads. They stay
396 // registered and erroring until the migration is old news.
397 if (is_user_defined("restartao"))
398 error("\n\n`restartao` has been retired: use `restart ao` instead\n\n");
399 if (is_user_defined("no_compute"))
400 error("\n\n`no_compute` has been retired: use `restart read_only` instead\n\n");
401 if (is_user_defined("nv_factor"))
402 error("\n\n`nv_factor` has been retired: the virtual step-down is now `nv_extra`, "
403 "`nv_step` and `nv_its`\n\n");
404
405 // population analysis: known schemes, `none` alone, and all-electron only (the
406 // projected basis functions describe all electrons of an atom)
407 {
408 const std::vector<std::string> pop = population();
409 bool any = false, none = false;
410 for (const auto& s : pop) {
411 MADNESS_CHECK_THROW(s == "none" or s == "mulliken" or s == "lowdin" or s == "iao",
412 "population: the schemes are none, mulliken, lowdin and iao");
413 any = any or s != "none";
414 none = none or s == "none";
415 }
416 MADNESS_CHECK_THROW(not (any and none), "population: none cannot be combined with a scheme");
417 MADNESS_CHECK_THROW(not any or (get<bool>("pure_ae") and not get<bool>("psp_calc") and n_core == 0),
418 "population: needs an all-electron calculation (no pseudopotentials or core potentials)");
419 set_derived_value("population_basis", aobasis());
420 }
421
422 // dispersion correction
423 if (dispersion()!="none" and xc()!="hf") {
424 set_derived_value("dispersion_functional",xc());
425 }
426
427
428 }
