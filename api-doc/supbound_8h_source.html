4/// \file supbound.h
5/// \brief Bounds on the largest magnitude of a function inside a box, from its scaling coefficients.
6///
7/// Inside the box (n, l) a function with scaling coefficients c is, in user coordinates,
8///
9/// f(x) = 2^{NDIM n/2} V^{-1/2} sum_i c_i prod_d phi_{i_d}(2^n x_d - l_d)
10///
11/// (x in simulation coordinates, V the cell volume; FunctionCommonData::coeffs2values), and the normalized
12/// Legendre scaling functions phi_i(t) = sqrt(2i+1) P_i(2t-1) peak at sqrt(2i+1) on the box faces. Hence
13///
14/// sup_box |f| <= 2^{NDIM n/2} V^{-1/2} sum_i |c_i| prod_d sqrt(2 i_d + 1),
15///
16/// rigorously, at k^NDIM multiply-adds per box. The bound lies above the box's rms magnitude
17/// ||c|| / sqrt(V_box) by a factor between 1 (a constant) and k^NDIM; for a function that is smooth across the
18/// box it is close to 1, and it is the larger the more f varies inside the box.
19///
20/// SupBoundMap holds that bound for the boxes of a tree: a leaf's from its coefficients, an interior node's as
21/// the max over its children (the sup over a union is the max of the sups), and a box below a leaf takes the
22/// leaf's (the sup over a sub-box is at most the sup over the box). An interior node's own coefficients (a
23/// redundant tree) are a projection of f and bound nothing, so only the leaves' coefficients are read.
24/// FunctionImpl::sup_bound_map() builds the map for a function.

36 /// Bounds on sup|f| per box of a function's tree (supbound.h)
37 template <std::size_t NDIM>
38 class SupBoundMap {
39 public:
40 typedef std::unordered_map<Key<NDIM>, double, Hash<Key<NDIM>>> mapT;
41
42 /// the answer for a box the map holds no bound for
43 static constexpr double unbounded() { return 1e300; }

45 /// the weights of the leaf bound, prod_d sqrt(2 i_d + 1), for k coefficients per dimension
46 static Tensor<double> leaf_weights(const int k) {
47 Tensor<double> w1(k);
48 for (int i = 0; i < k; ++i) w1(i) = std::sqrt(2.0 * i + 1.0);
49 Tensor<double> w = w1;
50 for (std::size_t d = 1; d < NDIM; ++d) w = outer(w, w1);
51 return w;
52 }
53
54 /// A bound on sup|f| over the box `key`, from f's scaling coefficients `c` there and the weights of
55 /// leaf_weights(k). No coefficients means f vanishes in the box.
56 template <typename T>
57 static double leaf_bound(const Key<NDIM>& key, const Tensor<T>& c, const Tensor<double>& weights) {
58 if (c.size() == 0) return 0.0;
59 MADNESS_CHECK_THROW(c.size() == weights.size(), "SupBoundMap: coefficients and weights differ in size");
60 const Tensor<T> cc = c.iscontiguous() ? c : copy(c);
61 const T* p = cc.ptr();
62 const double* w = weights.ptr();
63 double s = 0.0;
64 for (long i = 0; i < cc.size(); ++i) s += std::abs(p[i]) * w[i];
65 return s * std::pow(2.0, 0.5 * NDIM * key.level()) / std::sqrt(FunctionDefaults<NDIM>::get_cell_volume());
66 }

68 SupBoundMap() = default;
69 explicit SupBoundMap(mapT bound) : bound_(std::move(bound)) {}
70
71 /// The bound for the box `key`: its own, or the leaf's it lies below; unbounded() for a box the map
72 /// holds neither for. `below_leaf`, if given, is set when the answer is a leaf's above `key`.
73 double operator()(Key<NDIM> key, bool* below_leaf = nullptr) const {
74 for (bool below = false;; below = true) {
75 const auto it = bound_.find(key);
76 if (it != bound_.end()) {
77 if (below_leaf) *below_leaf = below;
78 return it->second;
79 }
80 if (key.level() == 0) return unbounded();
81 key = key.parent();
82 }
83 }

85 /// the number of boxes held
