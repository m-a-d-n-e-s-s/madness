1/*
2 This file is part of MADNESS.
3
4 Unified adaptive NS-form product/inner operator.
5
6 Generalises three pre-existing operators in funcimpl.h:
7 * multiply_op<LDIM> : h(1,2) = f(1,2) * g(p)
8 * Vphi_op_NS<opT,LDIM> : result = sum_t product_t (sum of single-multiplications)
9 * Vphi_ij_u_op_NS<opT,LDIM> : u * i(1) * j(2) and <eri | u*i*j>
10
11 A composite_term carries a ket (or a hartree pair p1*p2), a chain of LDIM factors per
12 particle, and an optional NDIM "extra" factor (typically an on-demand operator such as
13 an ERI). The result is sum over terms of the products. An inner-product mode contracts
14 each term against an NDIM "bra" function and accumulates a scalar at key0.
15
16 Error control follows Vphi_ij_u_op_NS::compute_error_from_inaccurate_refinement:
17 (s,d)-norm cross-terms over {ket, factors_1, factors_2, extra} summed across the
18 multiplication chain, plus tnorm/oversampling error from each pointwise_multiplier.
19*/

27/// One term of a composite product: ket(1,2) * factor1(1) * factor2(2) * extra(1,2)
28///
29/// The ket is either an explicit NDIM Function (`ket`) or a low-rank sum of hartree
30/// products ket(1,2) = Σ_k p1[k](1) ⊗ p2[k](2). Provide `ket` *or* (p1, p2); not both.
31template <typename T, std::size_t NDIM>
32struct composite_term {
33 static constexpr std::size_t LDIM = NDIM / 2;
34 static_assert(NDIM == 2 * LDIM, "composite_term requires NDIM = 2*LDIM");
35
36 using FN = Function<T, NDIM>;
37 using FL = Function<T, LDIM>;
38
39 /// NDIM ket factor. If uninitialized, the term ket is Σ_k p1[k] ⊗ p2[k].
40 FN ket;
41 /// parallel vectors forming the ket as a sum of hartree products (p1.size() == p2.size())
42 std::vector<FL> p1, p2;
43 /// single LDIM factor per particle (uninitialized = skip)
44 FL factor1, factor2;
45 /// optional single NDIM extra factor (e.g. an on-demand ERI). Multiplied into the
46 /// chain in function mode; ignored in inner mode (use the global `bra` instead).
47 FN extra;
48 /// scalar prefactor applied to the entire product. Used by the DSL to fold scaled
49 /// subtrees (`α * f`) into a single term without an explicit scalar Function literal.
50 double coeff = 1.0;
51};

128 /// retrieve scaling- and detail-norm at the given key from a CoeffTracker
129 template <typename CT, typename KEY>
130 static std::pair<double, double> sd_norm(const CT& ct, const KEY& key) {
131 if (!ct.get_impl()) return {1.0, 0.0};
132 return {ct.coeff(key).normf(), ct.dnorm(key)};
133 }
134
135 /// scaling- and detail-norm for an on-demand NDIM extra factor. Returns
136 /// (snorm, dnorm) ~ (||P_n eri||, ||Q_n eri||), with dnorm computed from the
137 /// NS s0 / non-s0 split of the eri coeff tensor.
138 static std::pair<double, double> sd_norm_extra(const implT* extra, const keyT& key) {
139 if (!extra) return {1.0, 0.0};
140 tensorT c = eri_coeffs(extra, key);
141 const double sn = c(extra->get_cdata().s0).normf();
142 tensorT d = copy(c);
143 d(extra->get_cdata().s0) = 0.0;
144 return {sn, d.normf()};
145 }
146
147 /// Σ over all non-trivial cross-products of (s_i + d_i) − ∏ s_i.
148 /// Each factor contributes (s,d); error = total − product_of_s.
149 static double cross_term_error(const std::vector<std::pair<double, double>>& sd) {
150 double total = 1.0, sprod = 1.0;
151 for (const auto& [s, d] : sd) { total *= (s + d); sprod *= s; }
152 return total - sprod;
153 }
154
155 /// (s,d) of a sum-of-hartree-products ket: snorm = Σ_k s1_k s2_k,
156 /// dnorm = Σ_k (s1_k d2_k + d1_k s2_k + d1_k d2_k). Upper-bound on the true
157 /// (s,d) of the rank-r sum.
158 std::pair<double, double> sd_norm_pair_sum(const std::vector<ctL>& iap1,
159 const std::vector<ctL>& iap2,
160 const keyL& k1, const keyL& k2) const {
161 MADNESS_ASSERT(iap1.size() == iap2.size());
162 double s = 0.0, d = 0.0;
163 for (std::size_t k = 0; k < iap1.size(); ++k) {
164 const auto [s1, d1] = sd_norm(iap1[k], k1);
165 const auto [s2, d2] = sd_norm(iap2[k], k2);
166 s += s1 * s2;
167 d += s1 * d2 + d1 * s2 + d1 * d2;
168 }
169 return {s, d};
170 }
171
172 /// full inaccurate-refinement error for one term, scaled by the term's coeff.
173 double term_refinement_error(const TermImpl& t, const keyT& key,
174 const keyL& k1, const keyL& k2) const {
175 std::vector<std::pair<double, double>> sd;
176 if (t.have_ket()) sd.push_back(sd_norm(t.iaket, key));
177 else sd.push_back(sd_norm_pair_sum(t.iap1, t.iap2, k1, k2));
178 if (t.have_factor1()) sd.push_back(sd_norm(t.ifactor1, k1));
179 if (t.have_factor2()) sd.push_back(sd_norm(t.ifactor2, k2));
180 if (t.have_extra()) sd.push_back(sd_norm_extra(t.extra, key));
181 return std::abs(t.coeff) * cross_term_error(sd);
182 }

188 /// fetch coeffs of an on-demand NDIM function at `key`
189 static tensorT eri_coeffs(const implT* eri, const keyT& key) {
190 MADNESS_ASSERT(eri);
191 if (eri->get_functor()->provides_coeff()) {
192 return eri->get_functor()->coeff(key).full_tensor();
193 } else {
194 tensorT val(eri->get_cdata().vk);
195 eri->fcube(key, *(eri->get_functor()), eri->get_cdata().quad_x, val);
196 return eri->values2coeffs(key, val);
197 }
198 }

200 /// build the NDIM ket coefficients at `key` for a term: either the explicit NDIM ket
201 /// or the sum Σ_k outer(p1[k](k1), p2[k](k2)).
202 coeffT make_term_ket(const TermImpl& t, const keyT& key,
203 const keyL& k1, const keyL& k2) const {
204 if (t.have_ket()) return t.iaket.coeff(key);
205 MADNESS_ASSERT(!t.iap1.empty() && t.iap1.size() == t.iap2.size());
206 coeffT acc = outer(t.iap1[0].coeff(k1), t.iap2[0].coeff(k2),
207 result->get_tensor_args());
208 for (std::size_t k = 1; k < t.iap1.size(); ++k) {
209 acc += outer(t.iap1[k].coeff(k1), t.iap2[k].coeff(k2),
210 result->get_tensor_args());
211 }
212 acc.reduce_rank(result->get_tensor_args().thresh);
213 return acc;
214 }

216 /// multiply one term: ket * factor1(1) * factor2(2) * extra(1,2)
217 /// Returns (coeffs at key, accumulated tnorm/oversampling error).
218 std::pair<coeffT, double> make_term_coeffs(const TermImpl& t, const keyT& key) const {
219 keyL k1, k2; key.break_apart(k1, k2);
220 coeffT current = make_term_ket(t, key, k1, k2);
221 double err = 0.0;
222
223 if (t.have_factor1()) {
224 typename implT::template pointwise_multiplier<LDIM> pm(key, current);
225 pm.oversampling = oversampling;
226 current = pm(key, t.ifactor1.coeff(k1).full_tensor(), 1);
227 err += pm.error;
228 }
229 if (t.have_factor2()) {
230 typename implT::template pointwise_multiplier<LDIM> pm(key, current);
231 pm.oversampling = oversampling;
232 current = pm(key, t.ifactor2.coeff(k2).full_tensor(), 2);
233 err += pm.error;
234 }
235 if (t.have_extra()) {
236 tensorT ceri = eri_coeffs(t.extra, key);
237 typename implT::template pointwise_multiplier<LDIM> pm(key, current);
238 pm.oversampling = oversampling;
239 tensorT v = pm(key, ceri);
240 current = coeffT(v, result->get_tensor_args());
241 err += pm.error;
242 }
243 // DSL scalar prefactor: scale coeffs and error
244 if (t.coeff != 1.0) {
245 current.scale(t.coeff);
246 err *= std::abs(t.coeff);
247 }
248 return {current, err};
249 }

251 /// Σ over terms. Function mode: returns the sum of NDIM coeffs.
252 /// Inner mode: returns a degenerate coeff (the scalar lives elsewhere);
253 /// here we return the sum of contractions packed as the
254 /// (0,…) element of a vk-shaped tensor.
255 std::pair<coeffT, double> make_sum_coeffs(const keyT& key) const {
256 keyL k1, k2; key.break_apart(k1, k2);
257
258 double error = 0.0;
259 if (mode == Mode::function) {
260 coeffT cresult(result->get_cdata().vk, result->get_tensor_args());
261 for (const auto& t : terms) {
262 error += term_refinement_error(t, key, k1, k2);
263 auto [c, e] = make_term_coeffs(t, key);
264 cresult += c;
265 error += e;
266 }
267 if (terms.size() > 1) cresult.reduce_rank(result->get_tensor_args().thresh);
268 return {cresult, error};
269 } else { // inner mode
270 // Fetch the bra coefficients at this key. Two paths:
271 // - on-demand bra: evaluate the functor / use provided coeffs
272 // - explicit-NDIM bra: read from the CoeffTracker (parent-to-child)
273 const tensorT cbra_t = have_explicit_bra()
274 ? cbra.coeff(key).full_tensor_copy()
275 : eri_coeffs(bra, key);
276 T sum = T(0);
277 for (const auto& t : terms) {
278 error += term_refinement_error(t, key, k1, k2);
279 auto [c, e] = make_term_coeffs(t, key);
280 error += e;
281 sum += cbra_t.trace(c.full_tensor_copy());
282 }
283 // wrap scalar in a vk tensor at key0 convention
284 tensorT scalar(result->get_cdata().vk);
285 scalar.flat()(0L) = sum;
286 return {coeffT(scalar, TensorArgs(TT_FULL, -1.0)), error};
287 }
288 }

307 // ---------------------------------------------------------------------
308 // tree traversal — mirrors Vphi_ij_u_op_NS::operator()
309 // ---------------------------------------------------------------------
310
311 std::pair<bool, coeffT> operator()(const keyT& key) const {
312 MADNESS_ASSERT(result->get_coeffs().is_local(key));
313
314 if (leaf_op.do_pre_screening()) {
315 if (leaf_op.pre_screening(key)) {
316 auto [c, err] = make_sum_coeffs(key);
317 leaf_action(key, c);
318 return {true, coeffT()};
319 }
320 return continue_recursion(std::vector<bool>(1 << NDIM, false), tensorT(), key);
321 }
322
323 const size_t il = result->get_initial_level();
324 const int min_lvl = FunctionDefaults<NDIM>::get_refine() ? int(il) + 1 : int(il);
325 if (key.level() < min_lvl)
326 return continue_recursion(std::vector<bool>(1 << NDIM, false), tensorT(), key);
327
328 if (key.level() < result->get_special_level() &&
329 leaf_op.special_refinement_needed(key))
330 return continue_recursion(std::vector<bool>(1 << NDIM, false), tensorT(), key);
331
332 auto [c, err] = make_sum_coeffs(key);
333
334 const double tol = result->truncate_tol(target_precision, key);
335 if (leaf_op.post_screening(key, c) || err < tol) {
336 leaf_action(key, c);
337 return {true, coeffT()};
338 }
339 return continue_recursion(std::vector<bool>(1 << NDIM, false), tensorT(), key);
340 }
341
342 std::pair<bool, coeffT> continue_recursion(const std::vector<bool> child_is_leaf,
343 const tensorT& coeffs,
344 const keyT& key) const {
345 std::size_t i = 0;
346 for (KeyChildIterator<NDIM> kit(key); kit; ++kit, ++i) {
347 const keyT child = kit.key();
348 if (child_is_leaf[i]) {
349 insert_op<T, NDIM> iop(result);
350 iop(child, coeffT(copy(coeffs(result->child_patch(child))),
351 result->get_tensor_args()), true);
352 } else {
353 this_type child_op = this->make_child(child);
354 noop<T, NDIM> no;
355 const ProcessID p = result->get_coeffs().owner(child);
356 void (implT::*ft)(const this_type&, const noop<T, NDIM>&, const keyT&) const
357 = &implT::template forward_traverse<this_type, noop<T, NDIM>>;
358 result->task(p, ft, child_op, no, child);
359 }
360 }
361 return {true, coeffT()};
362 }
363
364 this_type make_child(const keyT& child) const {
365 keyL k1, k2; child.break_apart(k1, k2);
366 std::vector<TermImpl> child_terms;
367 child_terms.reserve(terms.size());
368 for (const auto& t : terms) {
369 TermImpl ct;
370 ct.iaket = t.have_ket() ? t.iaket.make_child(child) : ctT();
371 if (!t.have_ket()) {
372 ct.iap1.reserve(t.iap1.size());
373 ct.iap2.reserve(t.iap2.size());
374 for (const auto& p : t.iap1) ct.iap1.push_back(p.make_child(k1));
375 for (const auto& p : t.iap2) ct.iap2.push_back(p.make_child(k2));
376 }
377 ct.ifactor1 = t.have_factor1() ? t.ifactor1.make_child(k1) : ctL();
378 ct.ifactor2 = t.have_factor2() ? t.ifactor2.make_child(k2) : ctL();
379 ct.extra = t.extra;
380 ct.coeff = t.coeff;
381 child_terms.push_back(std::move(ct));
382 }
383 ctT child_cbra = have_explicit_bra() ? cbra.make_child(child) : ctT{};
384 return this_type(result, leaf_op, child_terms, mode, bra,
385 target_precision, oversampling, child_cbra);
386 }

388 Future<this_type> activate() const {
389 // Collect all CoeffTracker activations into two flat future vectors without
390 // calling .get(). The forward_ctor task fires only once every future in
391 // these vectors is resolved, so .get() inside it is non-blocking.
392 // The old approach called .get() here, which entered the MADNESS work-
393 // stealing loop, picked up more forward_traverse tasks, and caused
394 // unbounded stack growth (stack overflow -> segfault).
395 //
396 // TaskFn supports at most 9 args, so we pack into two flat vectors:
397 // t_futs: [0..N-1] = ket futures (placeholder ctT{} if !have_ket), [N] = cbra
398 // l_futs: [0..p_total-1] = flat p1/p2 per term (p1[0..n-1], p2[0..n-1]),
399 // [p_total..p_total+N-1] = f1 futures,
400 // [p_total+N..p_total+2N-1] = f2 futures
401 const size_t N = terms.size();
402 std::vector<Future<ctT>> t_futs(N + 1);
403 std::vector<Future<ctL>> l_futs;
404 for (size_t i = 0; i < N; ++i) {
405 const auto& t = terms[i];
406 if (t.have_ket()) {
407 t_futs[i] = t.iaket.activate();
408 } else {
409 t_futs[i] = Future<ctT>(ctT{});
410 for (const auto& p : t.iap1) l_futs.push_back(p.activate());
411 for (const auto& p : t.iap2) l_futs.push_back(p.activate());
412 }
413 }
414 for (size_t i = 0; i < N; ++i) {
415 const auto& t = terms[i];
416 l_futs.push_back(t.have_factor1() ? t.ifactor1.activate() : Future<ctL>(ctL{}));
417 }
418 for (size_t i = 0; i < N; ++i) {
419 const auto& t = terms[i];
420 l_futs.push_back(t.have_factor2() ? t.ifactor2.activate() : Future<ctL>(ctL{}));
421 }
422 t_futs[N] = have_explicit_bra() ? cbra.activate() : Future<ctT>(ctT{});
423 return result->world.taskq.add(
424 detail::wrap_mem_fn(*const_cast<this_type*>(this), &this_type::forward_ctor),
425 result, leaf_op, terms, mode, bra, target_precision, oversampling,
426 t_futs, l_futs);
427 }
428
429 this_type forward_ctor(implT* result1, const opT& leaf_op1,
430 const std::vector<TermImpl>& orig_terms, Mode mode1,
431 const implT* bra1, double target_precision1, int oversampling1,
432 const std::vector<Future<ctT>>& t_futs,
433 const std::vector<Future<ctL>>& l_futs) {
434 // All futures are resolved when this task fires — .get() is non-blocking.
435 auto& tf = const_cast<std::vector<Future<ctT>>&>(t_futs);
436 auto& lf = const_cast<std::vector<Future<ctL>>&>(l_futs);
437 const size_t N = orig_terms.size();
438 // p_total: offset where f1/f2 futures begin in l_futs
439 size_t p_total = 0;
440 for (size_t i = 0; i < N; ++i)
441 if (!orig_terms[i].have_ket()) p_total += 2 * orig_terms[i].iap1.size();
442 std::vector<TermImpl> activated(N);
443 size_t p_idx = 0;
444 for (size_t i = 0; i < N; ++i) {
445 const auto& t = orig_terms[i];
446 activated[i].extra = t.extra;
447 activated[i].coeff = t.coeff;
448 if (t.have_ket()) {
449 activated[i].iaket = tf[i].get();
450 } else {
451 const size_t np = t.iap1.size();
452 for (size_t k = 0; k < np; ++k)
453 activated[i].iap1.push_back(lf[p_idx + k].get());
454 for (size_t k = 0; k < np; ++k)
455 activated[i].iap2.push_back(lf[p_idx + np + k].get());
456 p_idx += 2 * np;
457 }
458 if (t.have_factor1()) activated[i].ifactor1 = lf[p_total + i].get();
459 if (t.have_factor2()) activated[i].ifactor2 = lf[p_total + N + i].get();
460 }
461 return this_type(result1, leaf_op1, activated, mode1, bra1,
462 target_precision1, oversampling1, tf[N].get());
463 }

477/// Bring every input of a composite_term into redundant state.
478template <typename T, std::size_t NDIM>
479void make_term_inputs_redundant(composite_term<T, NDIM>& term) {
480 if (term.ket.is_initialized()) term.ket.change_tree_state(redundant, false);
481 for (auto& p : term.p1) p.change_tree_state(redundant, false);
482 for (auto& p : term.p2) p.change_tree_state(redundant, false);
483 if (term.factor1.is_initialized()) term.factor1.change_tree_state(redundant, false);
484 if (term.factor2.is_initialized()) term.factor2.change_tree_state(redundant, false);
485 // extra is on-demand → no state change
486}

488/// Restore every input of a composite_term to reconstructed state.
489template <typename T, std::size_t NDIM>
490void make_term_inputs_reconstructed(composite_term<T, NDIM>& term) {
491 if (term.ket.is_initialized()) term.ket.change_tree_state(reconstructed, false);
492 for (auto& p : term.p1) p.change_tree_state(reconstructed, false);
493 for (auto& p : term.p2) p.change_tree_state(reconstructed, false);
494 if (term.factor1.is_initialized()) term.factor1.change_tree_state(reconstructed, false);
495 if (term.factor2.is_initialized()) term.factor2.change_tree_state(reconstructed, false);
496}

498/// Translate a composite_term<T,NDIM> into the op's CoeffTracker-based TermImpl.
499template <typename T, std::size_t NDIM, typename OP>
500typename OP::TermImpl make_term_impl(const composite_term<T, NDIM>& term) {
501 constexpr std::size_t LDIM = NDIM / 2;
502 typename OP::TermImpl x;
503 if (term.ket.is_initialized()) {
504 x.iaket = CoeffTracker<T, NDIM>(term.ket.get_impl().get());
505 } else {
506 MADNESS_ASSERT(!term.p1.empty() && term.p1.size() == term.p2.size());
507 x.iap1.reserve(term.p1.size());
508 x.iap2.reserve(term.p2.size());
509 for (const auto& p : term.p1) x.iap1.emplace_back(p.get_impl().get());
510 for (const auto& p : term.p2) x.iap2.emplace_back(p.get_impl().get());
511 }
512 if (term.factor1.is_initialized())
513 x.ifactor1 = CoeffTracker<T, LDIM>(term.factor1.get_impl().get());
514 if (term.factor2.is_initialized())
515 x.ifactor2 = CoeffTracker<T, LDIM>(term.factor2.get_impl().get());
516 if (term.extra.is_initialized()) x.extra = term.extra.get_impl().get();
517 x.coeff = term.coeff;
518 return x;
519}

521} // namespace detail

523/// build result = Σ_t product_t adaptively
524///
525/// All input functions are brought into redundant state (cheap from reconstructed).
526/// The result is left in reconstructed state on exit.
527template <typename T, std::size_t NDIM, typename leaf_opT>
528void composite_product_apply(FunctionImpl<T, NDIM>* result,
529 const leaf_opT& leaf_op,
530 std::vector<composite_term<T, NDIM>>& terms,
531 double target_precision,
532 int oversampling,
533 bool fence) {
534 constexpr std::size_t LDIM = NDIM / 2;
535 using OP = composite_product_op_NS<T, NDIM, leaf_opT, LDIM>;
536
537 World& world = result->world;
538 for (auto& t : terms) detail::make_term_inputs_redundant(t);
539 world.gop.fence();
540
541 if (target_precision == 0.0) target_precision = result->get_thresh();
542
543 std::vector<typename OP::TermImpl> impl_terms;
544 impl_terms.reserve(terms.size());
545 for (const auto& t : terms) {
546 impl_terms.push_back(detail::make_term_impl<T, NDIM, OP>(t));
547 }
548
549 OP coeff_op(result, leaf_op, impl_terms, OP::Mode::function,
550 /*bra*/ nullptr, target_precision, oversampling);
551 noop<T, NDIM> apply_op;
552
553 if (world.rank() == result->get_coeffs().owner(result->get_cdata().key0)) {
554 result->task(world.rank(),
555 &FunctionImpl<T, NDIM>::template forward_traverse<OP, noop<T, NDIM>>,
556 coeff_op, apply_op, result->get_cdata().key0);
557 }
558
559 result->set_tree_state(redundant_after_merge);
560 if (fence) {
561 world.gop.fence();
562 result->flo_unary_op_node_inplace(
563 typename FunctionImpl<T, NDIM>::do_consolidate_buffer(result->get_tensor_args(),false),
564 true);
565 result->sum_down(true);
566 result->set_tree_state(reconstructed);
567
568 // Restore input tree state so callers can keep using the same Function handles
569 // (e.g., to compare against the old multiply / make_Vphi paths that expect
570 // reconstructed inputs).
571 for (auto& t : terms) detail::make_term_inputs_reconstructed(t);
572 world.gop.fence();
573 }
574}

576/// compute <bra | Σ_t product_t> adaptively.
577///
578/// `bra` may be on-demand (e.g. an ERI functor) — the operator evaluates its coeffs
579/// per box via fcube — or a regular tree-resident NDIM function, in which case the
580/// bra is brought into `redundant` state and a CoeffTracker walks it. Returns the
581/// accumulated scalar.
582template <typename T, std::size_t NDIM, typename leaf_opT>
583T composite_inner_apply(World& world,
584 const FunctionImpl<T, NDIM>* like,
585 const leaf_opT& leaf_op,
586 std::vector<composite_term<T, NDIM>>& terms,
587 Function<T, NDIM>& bra_function,
588 double target_precision,
589 int oversampling) {
590 constexpr std::size_t LDIM = NDIM / 2;
591 using OP = composite_product_op_NS<T, NDIM, leaf_opT, LDIM>;
592
593 MADNESS_CHECK_THROW(bra_function.is_initialized(),
594 "composite_inner: bra Function must be initialised");
595
596 // Per-term `extra` factors *are* allowed in inner mode: the trace becomes the triple
597 // integral ⟨bra | ket·factor1·factor2·extra⟩. `make_term_coeffs` already multiplies
598 // the extra into the term coefficients before the contraction.
599 for (auto& t : terms) {
600 detail::make_term_inputs_redundant(t);
601 }
602
603 // Decide bra dispatch path
604 const FunctionImpl<T,NDIM>* bra_impl = bra_function.get_impl().get();
605 const bool on_demand_bra = bra_impl->is_on_demand();
606 typename OP::ctT cbra_tracker;
607 if (!on_demand_bra) {
608 bra_function.change_tree_state(redundant, false);
609 cbra_tracker = typename OP::ctT(bra_impl);
610 }
611 world.gop.fence();
612
613 // Build a same-shape accumulator tree (empty initially); the scalar accumulates at key0.
614 FunctionImpl<T, NDIM> acc(*like, like->get_pmap(), false);
615 if (target_precision == 0.0) target_precision = acc.get_thresh() * 0.01;
616
617 std::vector<typename OP::TermImpl> impl_terms;
618 impl_terms.reserve(terms.size());
619 for (const auto& t : terms) {
620 impl_terms.push_back(detail::make_term_impl<T, NDIM, OP>(t));
621 }
622
623 OP coeff_op(&acc, leaf_op, impl_terms, OP::Mode::inner,
624 /*bra ptr*/ on_demand_bra ? bra_impl : nullptr,
625 target_precision, oversampling,
626 cbra_tracker);
627 noop<T, NDIM> apply_op;
628
629 if (world.rank() == acc.get_coeffs().owner(acc.get_cdata().key0)) {
630 acc.task(world.rank(),
631 &FunctionImpl<T, NDIM>::template forward_traverse<OP, noop<T, NDIM>>,
632 coeff_op, apply_op, acc.get_cdata().key0);
633 }
634 world.gop.fence();
635
636 // consolidate the buffer at key0 and read the scalar
637 acc.flo_unary_op_node_inplace(
638 typename FunctionImpl<T, NDIM>::do_consolidate_buffer(acc.get_tensor_args(), false), true);
639
640 T scalar = T(0);
641 if (world.rank() == acc.get_coeffs().owner(acc.get_cdata().key0)) {
642 typename FunctionImpl<T, NDIM>::dcT::const_accessor a;
643 if (acc.get_coeffs().find(a, acc.get_cdata().key0))
644 scalar = a->second.coeff().full_tensor_copy().flat()(0L);
645 }
646 world.gop.sum(scalar);
647
648 // Restore input tree state for the caller.
649 for (auto& t : terms) detail::make_term_inputs_reconstructed(t);
650 if (!on_demand_bra) bra_function.change_tree_state(reconstructed, false);
651 world.gop.fence();
652
653 return scalar;
654}
